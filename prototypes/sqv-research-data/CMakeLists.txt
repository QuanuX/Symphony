cmake_minimum_required(VERSION 3.30)
project(SymphonySqvResearchPrototype LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)

add_library(sqv-prepared-batch STATIC src/prepared_batch.cpp src/flow.cpp)
target_include_directories(sqv-prepared-batch PUBLIC include)
target_compile_features(sqv-prepared-batch PUBLIC cxx_std_26)
set_target_properties(sqv-prepared-batch PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(sqv-prepared-batch PRIVATE -Wall -Wextra -Wpedantic -Werror)

if(BUILD_TESTING)
  add_executable(sqv-prepared-batch-test tests/prepared_batch_test.cpp)
  target_link_libraries(sqv-prepared-batch-test PRIVATE sqv-prepared-batch)
  target_link_libraries(sqv-prepared-batch-test PRIVATE Threads::Threads)
  target_compile_options(sqv-prepared-batch-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME sqv-prepared-batch COMMAND sqv-prepared-batch-test)
  add_executable(sqv-flow-test tests/flow_test.cpp)
  target_link_libraries(sqv-flow-test PRIVATE sqv-prepared-batch)
  target_compile_options(sqv-flow-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME sqv-flow COMMAND sqv-flow-test)
endif()
