cmake_minimum_required(VERSION 3.30)
project(symphony_agent_plugin VERSION 0.1.0 LANGUAGES CXX)
if(NOT UNIX)
  message(FATAL_ERROR "This Alpha supports macOS and Linux. Windows transport is not implemented.")
endif()
include(CheckCXXCompilerFlag)
check_cxx_compiler_flag("-std=c++2c" SYMPHONY_HAS_CPP26)
if(NOT SYMPHONY_HAS_CPP26)
  message(FATAL_ERROR "A compiler accepting the C++26 draft mode (-std=c++2c) is required")
endif()
# CMake's AppleClang dialect table can lag the actual compiler. Require the
# compiler's checked C++26 flag explicitly rather than silently downgrading.
add_compile_options(-std=c++2c)
add_library(symphony_transport src/transport.cpp)
target_include_directories(symphony_transport PUBLIC include)
target_compile_options(symphony_transport PRIVATE -Wall -Wextra -Wpedantic)
add_executable(symphony-mcp src/main.cpp)
target_include_directories(symphony-mcp PRIVATE third_party)
target_link_libraries(symphony-mcp PRIVATE symphony_transport)
target_compile_options(symphony-mcp PRIVATE -Wall -Wextra -Wpedantic)
if(NOT APPLE)
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  target_link_libraries(symphony-mcp PRIVATE OpenSSL::Crypto)
endif()
option(SYMPHONY_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(SYMPHONY_SANITIZERS)
  target_compile_options(symphony_transport PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(symphony_transport PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(transport-test tests/transport_test.cpp)
  target_link_libraries(transport-test PRIVATE symphony_transport)
  add_test(NAME transport COMMAND transport-test)
  add_executable(hash-test tests/hash_test.cpp)
  target_include_directories(hash-test PRIVATE include)
  if(NOT APPLE)
    target_link_libraries(hash-test PRIVATE OpenSSL::Crypto)
  endif()
  add_test(NAME bounded-hashing COMMAND hash-test)
  add_test(NAME mcp-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mcp.py $<TARGET_FILE:symphony-mcp>)
  add_test(NAME client-adapters COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_adapters.py)
  set_tests_properties(transport mcp-integration PROPERTIES TIMEOUT 90)
endif()
install(TARGETS symphony-mcp RUNTIME DESTINATION bin)
