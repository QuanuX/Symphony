cmake_minimum_required(VERSION 3.25)
project(SymphonyScvGraphDuckdbConnector VERSION 0.2.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
if(NOT CMAKE_INSTALL_LIBEXECDIR STREQUAL "libexec" OR NOT CMAKE_INSTALL_DATADIR STREQUAL "share")
    message(FATAL_ERROR "The connector requires exact libexec/share receipt layout")
endif()
get_filename_component(SYMPHONY_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${SYMPHONY_REPOSITORY_ROOT}/cmake/SymphonyInstallReceiptV2.cmake")
option(SYMPHONY_KVE_USE_INSTALLED "Use an installed exact compatible C++ foundation" OFF)
if(SYMPHONY_KVE_USE_INSTALLED)
    find_package(SymphonyKnowledgeVectorEngine 0.1 REQUIRED CONFIG)
elseif(NOT TARGET Symphony::KnowledgeVectorEngine)
    set(SYMPHONY_KVE_BUILD_TESTING OFF CACHE BOOL "Build foundation tests separately" FORCE)
    add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp"
        "${CMAKE_CURRENT_BINARY_DIR}/knowledge-vector-engine-cpp" EXCLUDE_FROM_ALL)
endif()
# Explicit pinned dependency input; no network acquisition or global database installation.
set(DUCKDB_ROOT "" CACHE PATH "Verified DuckDB1.5.5 headers, library, license and provenance directory")
set(DUCKDB_LIBRARY "" CACHE FILEPATH "Exact DuckDB1.5.5 shared library for the selected architecture")
if(NOT EXISTS "${DUCKDB_ROOT}/duckdb.h" OR NOT EXISTS "${DUCKDB_ROOT}/LICENSE" OR NOT EXISTS "${DUCKDB_LIBRARY}")
    message(FATAL_ERROR "Supply DUCKDB_ROOT and DUCKDB_LIBRARY for the exact verified DuckDB1.5.5 dependency")
endif()
if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
    message(FATAL_ERROR "This pinned dependency recipe is macos-amd64; supply a reviewed platform recipe before expanding package support")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/DUCKDB-PROVENANCE.json" dependency_provenance)
foreach(pair "duckdb.h|${DUCKDB_ROOT}/duckdb.h" "LICENSE|${DUCKDB_ROOT}/LICENSE" "x86_64/libduckdb.dylib|${DUCKDB_LIBRARY}")
    string(REPLACE "|" ";" parts "${pair}")
    list(GET parts 0 key)
    list(GET parts 1 path)
    string(JSON expected_hash GET "${dependency_provenance}" files "${key}" sha256)
    file(SHA256 "${path}" actual_hash)
    if(NOT actual_hash STREQUAL expected_hash)
        message(FATAL_ERROR "Pinned DuckDB dependency differs: ${key}")
    endif()
endforeach()
add_library(scv-pinned-duckdb SHARED IMPORTED)
set_target_properties(scv-pinned-duckdb PROPERTIES IMPORTED_LOCATION "${DUCKDB_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${DUCKDB_ROOT}")
find_package(Threads REQUIRED)
set(CONNECTOR_MODULE scv-graph-duckdb-connector)
set(CONNECTOR_EXECUTABLE symphony-scv-graph-duckdb-connector)
set(CONNECTOR_VERSION "${PROJECT_VERSION}-dev")
set(CONNECTOR_RUNTIME "libexec/symphony/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
set(CONNECTOR_DOCS "share/doc/symphony/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
set(CONNECTOR_SCHEMAS "share/symphony/schemas/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
set(CONNECTOR_LICENSES "share/licenses/symphony-${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
add_library(scv-graph-duckdb-core STATIC src/connector.cpp)
target_include_directories(scv-graph-duckdb-core PUBLIC src)
target_link_libraries(scv-graph-duckdb-core PUBLIC Symphony::KnowledgeVectorEngine scv-pinned-duckdb Threads::Threads)
add_executable(${CONNECTOR_EXECUTABLE} src/main.cpp)
if(APPLE)
    set_target_properties(${CONNECTOR_EXECUTABLE} PROPERTIES INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH OFF)
    set(DUCKDB_INSTALLED_NAME libduckdb.dylib)
else()
    set_target_properties(${CONNECTOR_EXECUTABLE} PROPERTIES INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH OFF)
    set(DUCKDB_INSTALLED_NAME libduckdb.so)
endif()
target_link_libraries(${CONNECTOR_EXECUTABLE} PRIVATE scv-graph-duckdb-core)
foreach(target scv-graph-duckdb-core ${CONNECTOR_EXECUTABLE})
    set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endforeach()
if(BUILD_TESTING)
    # The installed target never links these test barriers. This separate target
    # compiles the same connector implementation for deterministic SIGKILL tests.
    add_executable(scv-graph-duckdb-commit-fault-test EXCLUDE_FROM_ALL
        src/main.cpp src/connector.cpp tests/commit_barrier.cpp)
    target_compile_definitions(scv-graph-duckdb-commit-fault-test PRIVATE SYMPHONY_SCV_GRAPH_INDEX_TEST_BARRIERS=1)
    target_include_directories(scv-graph-duckdb-commit-fault-test PRIVATE src tests)
    target_link_libraries(scv-graph-duckdb-commit-fault-test PRIVATE Symphony::KnowledgeVectorEngine scv-pinned-duckdb Threads::Threads)
    set_target_properties(scv-graph-duckdb-commit-fault-test PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    target_compile_options(scv-graph-duckdb-commit-fault-test PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME scv-graph-duckdb-connector-process COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/connector_test.py" --engine $<TARGET_FILE:${CONNECTOR_EXECUTABLE}> --library "${DUCKDB_LIBRARY}" --version "${CONNECTOR_VERSION}")
    add_test(NAME scv-graph-duckdb-inventory COMMAND ${Python3_EXECUTABLE}
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/inventory_test.py" --engine $<TARGET_FILE:${CONNECTOR_EXECUTABLE}> --library "${DUCKDB_LIBRARY}")
endif()
# The installation declarations below are kept explicit and receipt owned.
set(CONNECTOR_SCHEMA_FILES "${CMAKE_CURRENT_SOURCE_DIR}/schemas/v2/graph-index.schema.json")
set(CONNECTOR_OWNED "${CONNECTOR_RUNTIME}/${CONNECTOR_EXECUTABLE}|executable" "${CONNECTOR_RUNTIME}/${DUCKDB_INSTALLED_NAME}|regular" "${CONNECTOR_DOCS}/DUCKDB-PROVENANCE.json|regular" "${CONNECTOR_LICENSES}/DuckDB-LICENSE.MIT|regular" "${CONNECTOR_DOCS}/GRAPH-INDEX.md|regular" "${CONNECTOR_DOCS}/INDEX-MAINTENANCE.md|regular")
set(CONNECTOR_INSTALLED "${CONNECTOR_RUNTIME}/${CONNECTOR_EXECUTABLE}" "${CONNECTOR_RUNTIME}/${DUCKDB_INSTALLED_NAME}" "${CONNECTOR_DOCS}/DUCKDB-PROVENANCE.json" "${CONNECTOR_LICENSES}/DuckDB-LICENSE.MIT" "${CONNECTOR_DOCS}/GRAPH-INDEX.md" "${CONNECTOR_DOCS}/INDEX-MAINTENANCE.md")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "share/symphony/receipts/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}/install-receipt.json")
foreach(doc INTENT.md MANIFEST.md INSTALL.md SKILL.md SPEC.md FEATURES.md)
    list(APPEND CONNECTOR_OWNED "${CONNECTOR_DOCS}/${doc}|regular")
    list(APPEND CONNECTOR_INSTALLED "${CONNECTOR_DOCS}/${doc}")
    install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" DESTINATION "${CONNECTOR_DOCS}")
endforeach()
foreach(schema IN LISTS CONNECTOR_SCHEMA_FILES)
    get_filename_component(name "${schema}" NAME)
    list(APPEND CONNECTOR_OWNED "${CONNECTOR_SCHEMAS}/${name}|regular")
    list(APPEND CONNECTOR_INSTALLED "${CONNECTOR_SCHEMAS}/${name}")
endforeach()
list(APPEND CONNECTOR_OWNED "${CONNECTOR_LICENSES}/LICENSE-AGPL-3.0|regular" "${CONNECTOR_LICENSES}/nlohmann-json-LICENSE.MIT|regular")
list(APPEND CONNECTOR_INSTALLED "${CONNECTOR_LICENSES}/LICENSE-AGPL-3.0" "${CONNECTOR_LICENSES}/nlohmann-json-LICENSE.MIT")
install(FILES "${DUCKDB_LIBRARY}" DESTINATION "${CONNECTOR_RUNTIME}" RENAME "${DUCKDB_INSTALLED_NAME}")
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/DUCKDB-PROVENANCE.json" DESTINATION "${CONNECTOR_DOCS}")
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/knowledge/scv/GRAPH-INDEX.md" "${SYMPHONY_REPOSITORY_ROOT}/knowledge/scv/INDEX-MAINTENANCE.md" DESTINATION "${CONNECTOR_DOCS}")
install(FILES "${DUCKDB_ROOT}/LICENSE" DESTINATION "${CONNECTOR_LICENSES}" RENAME DuckDB-LICENSE.MIT)
install(TARGETS ${CONNECTOR_EXECUTABLE} RUNTIME DESTINATION "${CONNECTOR_RUNTIME}")
install(FILES ${CONNECTOR_SCHEMA_FILES} DESTINATION "${CONNECTOR_SCHEMAS}")
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/LICENSE" DESTINATION "${CONNECTOR_LICENSES}" RENAME LICENSE-AGPL-3.0)
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT"
    DESTINATION "${CONNECTOR_LICENSES}" RENAME nlohmann-json-LICENSE.MIT)
symphony_install_receipt_v2(COMPONENT_ID "${CONNECTOR_MODULE}" COMPONENT_KIND adapter
    MODULE_ID "${CONNECTOR_MODULE}" VECTOR_ID scv ENGINE_ID "${CONNECTOR_EXECUTABLE}"
    PACKAGE_ID "${CONNECTOR_MODULE}" VERSION "${CONNECTOR_VERSION}"
    RECEIPT_PATH "share/symphony/receipts/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}/install-receipt.json"
    OWNED_FILES ${CONNECTOR_OWNED}
    ENTRY_POINTS "${CONNECTOR_EXECUTABLE}|executable|${CONNECTOR_RUNTIME}/${CONNECTOR_EXECUTABLE}|symphony.knowledge.engine-process.v1")
configure_file(cmake/uninstall.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" @ONLY)
add_custom_target(uninstall-${CONNECTOR_MODULE} COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX}
    -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
