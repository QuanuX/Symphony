cmake_minimum_required(VERSION 3.30)
project(ScabvTws1045Conformance LANGUAGES CXX)
find_package(SymphonyScabvIbkrTws 0.1.0 EXACT REQUIRED CONFIG)
set(IBKR_SDK_INCLUDE_DIR "" CACHE PATH "Authorized SDK 10.45 source/cppclient/client directory")
set(IBKR_SDK_RELEASE "" CACHE STRING "Explicit externally verified SDK release")
if(NOT IBKR_SDK_RELEASE STREQUAL "10.45" OR NOT IS_ABSOLUTE "${IBKR_SDK_INCLUDE_DIR}")
 message(FATAL_ERROR "Supply an authorized, independently version-verified 10.45 SDK include directory and release. No download is performed.")
endif()
foreach(header EClient.h Contract.h)
 if(NOT EXISTS "${IBKR_SDK_INCLUDE_DIR}/${header}")
  message(FATAL_ERROR "Missing authorized SDK header: ${header}")
 endif()
endforeach()
add_library(sdk1045-conformance OBJECT conformance.cpp)
target_include_directories(sdk1045-conformance PRIVATE "${IBKR_SDK_INCLUDE_DIR}")
target_link_libraries(sdk1045-conformance PRIVATE Symphony::ScabvIbkrTws)
set_target_properties(sdk1045-conformance PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
