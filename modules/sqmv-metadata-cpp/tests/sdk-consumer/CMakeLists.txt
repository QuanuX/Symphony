cmake_minimum_required(VERSION 3.30)
project(IndependentInstalledSqmvMetadataConsumer LANGUAGES CXX)
find_package(SymphonySqmvMetadata 0.2.0 EXACT REQUIRED CONFIG)
if(NOT SymphonySqmvMetadata_RELEASE_VERSION STREQUAL "0.2.0-dev")
    message(FATAL_ERROR "unexpected exact SQMV release")
endif()
add_executable(consumer main.cpp)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(consumer PRIVATE Symphony::SqmvMetadata)
