cmake_minimum_required(VERSION 3.25)
get_filename_component(SYMPHONY_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${SYMPHONY_REPOSITORY_ROOT}/cmake/ShvPublicationInterface.generated.cmake")
project(SymphonyShvPublicationEngine VERSION ${SHV_OWNER_VERSION} LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
if(NOT CMAKE_INSTALL_LIBEXECDIR STREQUAL "libexec" OR NOT CMAKE_INSTALL_DATADIR STREQUAL "share")
 message(FATAL_ERROR "SHV requires exact libexec/share receipt layout")
endif()
get_filename_component(SYMPHONY_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${SYMPHONY_REPOSITORY_ROOT}/cmake/SymphonyInstallReceiptV2.cmake")
option(SYMPHONY_KVE_USE_INSTALLED "Use independently installed foundation" OFF)
if(SYMPHONY_KVE_USE_INSTALLED)
 find_package(SymphonyKnowledgeVectorEngine 0.1.0 EXACT REQUIRED CONFIG)
elseif(NOT TARGET Symphony::KnowledgeVectorEngine)
 set(SYMPHONY_KVE_BUILD_TESTING OFF CACHE BOOL "Foundation tests run separately" FORCE)
 add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/knowledge-vector-engine-cpp" EXCLUDE_FROM_ALL)
endif()
add_library(shv-publication-core STATIC src/publication.cpp src/descriptor.cpp ../shv-partition-engine/src/partition.cpp ../shv-partition-engine/src/descriptor.cpp)
target_include_directories(shv-publication-core PRIVATE ../shv-partition-engine/src)
target_include_directories(shv-publication-core PUBLIC src)
target_link_libraries(shv-publication-core PUBLIC Symphony::KnowledgeVectorEngine)
add_executable(symphony-shv-publication src/main.cpp)
target_link_libraries(symphony-shv-publication PRIVATE shv-publication-core)
foreach(target shv-publication-core symphony-shv-publication)
 set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endforeach()
if(BUILD_TESTING)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME shv-publication-interface COMMAND ${Python3_EXECUTABLE} "${SYMPHONY_REPOSITORY_ROOT}/tools/shv-interface-codegen/tests/interface_test.py" --owner shv-publication-engine --engine $<TARGET_FILE:symphony-shv-publication>)
 add_test(NAME shv-publication-process COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/publication_test.py" --engine $<TARGET_FILE:symphony-shv-publication>)
endif()
set(SHV_RUNTIME "libexec/symphony/shv-publication-engine/${PROJECT_VERSION}-dev")
set(SHV_DOCS "share/doc/symphony/shv-publication-engine/${PROJECT_VERSION}-dev")
set(SHV_SCHEMAS "share/symphony/schemas/shv-publication-engine/${PROJECT_VERSION}-dev")
set(SHV_LICENSES "share/licenses/symphony-shv-publication-engine/${PROJECT_VERSION}-dev")
set(SHV_OWNED "${SHV_RUNTIME}/symphony-shv-publication|executable")
set(SHV_INSTALLED "${SHV_RUNTIME}/symphony-shv-publication")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "share/symphony/receipts/shv-publication-engine/${PROJECT_VERSION}-dev/install-receipt.json")
install(TARGETS symphony-shv-publication RUNTIME DESTINATION "${SHV_RUNTIME}")
foreach(doc INTENT.md MANIFEST.md SPEC.md SKILL.md INSTALL.md FEATURES.md)
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" DESTINATION "${SHV_DOCS}")
 list(APPEND SHV_OWNED "${SHV_DOCS}/${doc}|regular")
 list(APPEND SHV_INSTALLED "${SHV_DOCS}/${doc}")
endforeach()
foreach(path IN LISTS SHV_OWNER_COMPANIONS)
 get_filename_component(name "${path}" NAME)
 install(FILES "${path}" DESTINATION "${SHV_DOCS}")
 list(APPEND SHV_OWNED "${SHV_DOCS}/${name}|regular")
 list(APPEND SHV_INSTALLED "${SHV_DOCS}/${name}")
endforeach()
foreach(path IN LISTS SHV_OWNER_SCHEMA_FILES)
 get_filename_component(name "${path}" NAME)
 install(FILES "${path}" DESTINATION "${SHV_SCHEMAS}")
 list(APPEND SHV_OWNED "${SHV_SCHEMAS}/${name}|regular")
 list(APPEND SHV_INSTALLED "${SHV_SCHEMAS}/${name}")
endforeach()
set(SHV_INTERFACE "share/symphony/contracts/shv-publication-engine/${PROJECT_VERSION}-dev/OWNER-INTERFACE.json")
get_filename_component(SHV_INTERFACE_DIR "${SHV_INTERFACE}" DIRECTORY)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/OWNER-INTERFACE.json" DESTINATION "${SHV_INTERFACE_DIR}")
list(APPEND SHV_OWNED "${SHV_INTERFACE}|regular")
list(APPEND SHV_INSTALLED "${SHV_INTERFACE}")
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/LICENSE" DESTINATION "${SHV_LICENSES}" RENAME LICENSE-AGPL-3.0)
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT" DESTINATION "${SHV_LICENSES}" RENAME nlohmann-json-LICENSE.MIT)
foreach(name LICENSE-AGPL-3.0 nlohmann-json-LICENSE.MIT)
 list(APPEND SHV_OWNED "${SHV_LICENSES}/${name}|regular")
 list(APPEND SHV_INSTALLED "${SHV_LICENSES}/${name}")
endforeach()
symphony_install_receipt_v2(COMPONENT_ID shv-publication-engine COMPONENT_KIND vector_engine MODULE_ID shv-publication-engine VECTOR_ID shv ENGINE_ID symphony-shv-publication PACKAGE_ID shv-publication-engine VERSION ${PROJECT_VERSION}-dev
 RECEIPT_PATH "share/symphony/receipts/shv-publication-engine/${PROJECT_VERSION}-dev/install-receipt.json" OWNED_FILES ${SHV_OWNED}
 ENTRY_POINTS "symphony-shv-publication|executable|${SHV_RUNTIME}/symphony-shv-publication|symphony.knowledge.engine-process.v1")
configure_file(cmake/uninstall.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" @ONLY)
add_custom_target(uninstall-shv-publication-engine COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
