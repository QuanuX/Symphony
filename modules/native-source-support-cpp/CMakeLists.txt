cmake_minimum_required(VERSION 3.30)
project(SymphonyNativeSourceSupport VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${REPO}/cmake/SymphonyInstallReceiptV2.cmake")
if(CMAKE_CONFIGURATION_TYPES)
  message(FATAL_ERROR "Exact receipt exports require a single-configuration generator")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
string(TOLOWER "${CMAKE_BUILD_TYPE}" CONFIG_LOWER)
option(SYMPHONY_NATIVESOURCESUPPORT_INSTALLED "Use exact installed dependencies" OFF)
if(SYMPHONY_NATIVESOURCESUPPORT_INSTALLED)
  find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT REQUIRED CONFIG)
else()
  function(source_dependencies)
    set(BUILD_TESTING OFF)
    if(NOT TARGET Symphony::KnowledgeVectorEngine)
      add_subdirectory("${REPO}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/symphony-knowledge-vector-engine" EXCLUDE_FROM_ALL)
    endif()
  endfunction()
  source_dependencies()
endif()
find_package(CURL 8.7.1 EXACT REQUIRED)
set(RELEASE "${PROJECT_VERSION}-dev")
set(INCLUDE_ROOT "${CMAKE_INSTALL_INCLUDEDIR}/symphony/native-source-support-cpp/${RELEASE}")
set(CMAKE_DIR "${CMAKE_INSTALL_LIBDIR}/cmake/SymphonyNativeSourceSupport-${RELEASE}")
set(ARCHIVE_DIR "${CMAKE_INSTALL_LIBDIR}/symphony/native-source-support-cpp/${RELEASE}")
set(DOC_DIR "share/doc/symphony/native-source-support-cpp/${RELEASE}")
set(LICENSE_DIR "share/licenses/symphony-native-source-support-cpp/${RELEASE}")
set(RECEIPT "share/symphony/receipts/native-source-support-cpp/${RELEASE}/install-receipt.json")
if(APPLE)
  set(CMAKE_CXX_ARCHIVE_CREATE "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_AR> qc <TARGET> <LINK_FLAGS> <OBJECTS>")
  set(CMAKE_CXX_ARCHIVE_APPEND "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_AR> q <TARGET> <LINK_FLAGS> <OBJECTS>")
  set(CMAKE_CXX_ARCHIVE_FINISH "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_RANLIB> <TARGET>")
endif()
add_library(native-source-support STATIC src/support.cpp src/http.cpp)
add_library(Symphony::NativeSourceSupport ALIAS native-source-support)
target_link_libraries(native-source-support PUBLIC Symphony::KnowledgeVectorEngine CURL::libcurl)
set_target_properties(native-source-support PROPERTIES EXPORT_NAME NativeSourceSupport CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_features(native-source-support PUBLIC cxx_std_26)
target_include_directories(native-source-support PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${INCLUDE_ROOT}>)
target_compile_options(native-source-support PRIVATE -Wall -Wextra -Wpedantic -Werror)
symphony_install_receipt_v2_preflight(RECEIPT_PATH "${RECEIPT}")
set(OWNED "")
set(INSTALLED_FILES "")
macro(own path)
  list(APPEND OWNED "${path}|regular")
  list(APPEND INSTALLED_FILES "${path}")
endmacro()
install(TARGETS native-source-support EXPORT SymphonyNativeSourceSupportTargets ARCHIVE DESTINATION "${ARCHIVE_DIR}")
own("${ARCHIVE_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}native-source-support${CMAKE_STATIC_LIBRARY_SUFFIX}")
if(APPLE)
    install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 \"${CMAKE_RANLIB}\" \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${ARCHIVE_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}native-source-support${CMAKE_STATIC_LIBRARY_SUFFIX}\" COMMAND_ERROR_IS_FATAL ANY)")
endif()
install(FILES "include/symphony/source/support.hpp" DESTINATION "${INCLUDE_ROOT}/symphony/source")
own("${INCLUDE_ROOT}/symphony/source/support.hpp")
install(FILES "include/symphony/source/json.hpp" DESTINATION "${INCLUDE_ROOT}/symphony/source")
own("${INCLUDE_ROOT}/symphony/source/json.hpp")
configure_package_config_file(cmake/SymphonyNativeSourceSupportConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/SymphonyNativeSourceSupportConfig.cmake" INSTALL_DESTINATION "${CMAKE_DIR}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/SymphonyNativeSourceSupportConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(EXPORT SymphonyNativeSourceSupportTargets FILE SymphonyNativeSourceSupportTargets.cmake NAMESPACE Symphony:: DESTINATION "${CMAKE_DIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SymphonyNativeSourceSupportConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/SymphonyNativeSourceSupportConfigVersion.cmake" DESTINATION "${CMAKE_DIR}")
foreach(name SymphonyNativeSourceSupportConfig.cmake SymphonyNativeSourceSupportConfigVersion.cmake SymphonyNativeSourceSupportTargets.cmake "SymphonyNativeSourceSupportTargets-${CONFIG_LOWER}.cmake")
 own("${CMAKE_DIR}/${name}")
endforeach()
foreach(doc INTENT.md MANIFEST.md SPEC.md SKILL.md INSTALL.md FEATURES.md)
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" DESTINATION "${DOC_DIR}")
 own("${DOC_DIR}/${doc}")
endforeach()
install(FILES "${REPO}/LICENSE" DESTINATION "${LICENSE_DIR}" RENAME LICENSE-AGPL-3.0)
own("${LICENSE_DIR}/LICENSE-AGPL-3.0")
symphony_install_receipt_v2(COMPONENT_ID native-source-support-cpp COMPONENT_KIND module MODULE_ID native-source-support-cpp VECTOR_ID sqav PACKAGE_ID native-source-support-cpp VERSION "${RELEASE}" RECEIPT_PATH "${RECEIPT}" OWNED_FILES ${OWNED})
configure_file(cmake/uninstall.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" @ONLY)
add_custom_target(uninstall-native-source-support-cpp COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
if(BUILD_TESTING)
 add_executable(native-source-support-test tests/test.cpp)
 target_link_libraries(native-source-support-test PRIVATE Symphony::NativeSourceSupport)
 target_compile_options(native-source-support-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 set_target_properties(native-source-support-test PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 add_test(NAME native-source-support COMMAND native-source-support-test)
endif()

if(BUILD_TESTING)
 add_executable(native-source-http-test tests/http_test.cpp src/http.cpp)
 target_compile_definitions(native-source-http-test PRIVATE SYMPHONY_SOURCE_HTTP_TESTING=1)
 target_link_libraries(native-source-http-test PRIVATE Symphony::NativeSourceSupport)
 target_compile_options(native-source-http-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 set_target_properties(native-source-http-test PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 add_test(NAME native-source-http COMMAND native-source-http-test)
endif()
