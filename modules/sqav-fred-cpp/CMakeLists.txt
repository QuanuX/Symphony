cmake_minimum_required(VERSION 3.30)
project(SymphonySqavFred VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${REPO}/cmake/SymphonyInstallReceiptV2.cmake")
if(CMAKE_CONFIGURATION_TYPES)
  message(FATAL_ERROR "Exact receipt exports require a single-configuration generator")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
string(TOLOWER "${CMAKE_BUILD_TYPE}" CONFIG_LOWER)
option(SYMPHONY_SQAVFRED_INSTALLED "Use exact installed dependencies" OFF)
if(SYMPHONY_SQAVFRED_INSTALLED)
  find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT REQUIRED CONFIG)
  find_package(SymphonyNativeSourceSupport 0.1.0 EXACT REQUIRED CONFIG)
  find_package(SymphonySqavCapture 0.2.0 EXACT REQUIRED CONFIG)
else()
  function(source_dependencies)
    set(BUILD_TESTING OFF)
    if(NOT TARGET Symphony::KnowledgeVectorEngine)
      add_subdirectory("${REPO}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/symphony-knowledge-vector-engine" EXCLUDE_FROM_ALL)
    endif()
    if(NOT TARGET Symphony::NativeSourceSupport)
      add_subdirectory("${REPO}/modules/native-source-support-cpp" "${CMAKE_CURRENT_BINARY_DIR}/native-source-support" EXCLUDE_FROM_ALL)
    endif()
    if(NOT TARGET Symphony::SqavCapture)
      add_subdirectory("${REPO}/modules/sqav-capture-cpp" "${CMAKE_CURRENT_BINARY_DIR}/sqav-capture" EXCLUDE_FROM_ALL)
    endif()
  endfunction()
  source_dependencies()
endif()

set(RELEASE "${PROJECT_VERSION}-dev")
set(INCLUDE_ROOT "${CMAKE_INSTALL_INCLUDEDIR}/symphony/sqav-fred-cpp/${RELEASE}")
set(CMAKE_DIR "${CMAKE_INSTALL_LIBDIR}/cmake/SymphonySqavFred-${RELEASE}")
set(ARCHIVE_DIR "${CMAKE_INSTALL_LIBDIR}/symphony/sqav-fred-cpp/${RELEASE}")
set(DOC_DIR "share/doc/symphony/sqav-fred-cpp/${RELEASE}")
set(LICENSE_DIR "share/licenses/symphony-sqav-fred-cpp/${RELEASE}")
set(RECEIPT "share/symphony/receipts/sqav-fred-cpp/${RELEASE}/install-receipt.json")
if(APPLE)
  set(CMAKE_CXX_ARCHIVE_CREATE "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_AR> qc <TARGET> <LINK_FLAGS> <OBJECTS>")
  set(CMAKE_CXX_ARCHIVE_APPEND "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_AR> q <TARGET> <LINK_FLAGS> <OBJECTS>")
  set(CMAKE_CXX_ARCHIVE_FINISH "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_RANLIB> <TARGET>")
endif()
add_library(sqav-fred STATIC src/fred.cpp)
add_library(Symphony::SqavFred ALIAS sqav-fred)
target_link_libraries(sqav-fred PUBLIC Symphony::KnowledgeVectorEngine Symphony::NativeSourceSupport Symphony::SqavCapture)
set_target_properties(sqav-fred PROPERTIES EXPORT_NAME SqavFred CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_features(sqav-fred PUBLIC cxx_std_26)
target_include_directories(sqav-fred PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${INCLUDE_ROOT}>)
target_compile_options(sqav-fred PRIVATE -Wall -Wextra -Wpedantic -Werror)
symphony_install_receipt_v2_preflight(RECEIPT_PATH "${RECEIPT}")
set(OWNED "")
set(INSTALLED_FILES "")
macro(own path)
  list(APPEND OWNED "${path}|regular")
  list(APPEND INSTALLED_FILES "${path}")
endmacro()
install(TARGETS sqav-fred EXPORT SymphonySqavFredTargets ARCHIVE DESTINATION "${ARCHIVE_DIR}")
own("${ARCHIVE_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}sqav-fred${CMAKE_STATIC_LIBRARY_SUFFIX}")
if(APPLE)
    install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 \"${CMAKE_RANLIB}\" \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${ARCHIVE_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}sqav-fred${CMAKE_STATIC_LIBRARY_SUFFIX}\" COMMAND_ERROR_IS_FATAL ANY)")
endif()
install(FILES "include/symphony/sqav/fred.hpp" DESTINATION "${INCLUDE_ROOT}/symphony/sqav")
own("${INCLUDE_ROOT}/symphony/sqav/fred.hpp")
configure_package_config_file(cmake/SymphonySqavFredConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/SymphonySqavFredConfig.cmake" INSTALL_DESTINATION "${CMAKE_DIR}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/SymphonySqavFredConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(EXPORT SymphonySqavFredTargets FILE SymphonySqavFredTargets.cmake NAMESPACE Symphony:: DESTINATION "${CMAKE_DIR}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SymphonySqavFredConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/SymphonySqavFredConfigVersion.cmake" DESTINATION "${CMAKE_DIR}")
foreach(name SymphonySqavFredConfig.cmake SymphonySqavFredConfigVersion.cmake SymphonySqavFredTargets.cmake "SymphonySqavFredTargets-${CONFIG_LOWER}.cmake")
 own("${CMAKE_DIR}/${name}")
endforeach()
foreach(doc INTENT.md MANIFEST.md SPEC.md SKILL.md INSTALL.md FEATURES.md)
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" DESTINATION "${DOC_DIR}")
 own("${DOC_DIR}/${doc}")
endforeach()
install(FILES "${REPO}/LICENSE" DESTINATION "${LICENSE_DIR}" RENAME LICENSE-AGPL-3.0)
own("${LICENSE_DIR}/LICENSE-AGPL-3.0")
symphony_install_receipt_v2(COMPONENT_ID sqav-fred-cpp COMPONENT_KIND module MODULE_ID sqav-fred-cpp VECTOR_ID sqav PACKAGE_ID sqav-fred-cpp VERSION "${RELEASE}" RECEIPT_PATH "${RECEIPT}" OWNED_FILES ${OWNED})
configure_file(cmake/uninstall.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" @ONLY)
add_custom_target(uninstall-sqav-fred-cpp COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
if(BUILD_TESTING)
 add_executable(sqav-fred-test tests/test.cpp)
 target_link_libraries(sqav-fred-test PRIVATE Symphony::SqavFred)
 target_compile_options(sqav-fred-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 set_target_properties(sqav-fred-test PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 add_test(NAME sqav-fred COMMAND sqav-fred-test)
endif()

if(BUILD_TESTING AND SQV_VERIFY_RETENTION)
 target_compile_definitions(sqav-fred-test PRIVATE SQV_VERIFY_RETENTION)
 target_include_directories(sqav-fred-test PRIVATE "${REPO}/tests/sqv-nonlive-sources")
 target_link_libraries(sqav-fred-test PRIVATE Symphony::SqdvDelivery)
endif()
