cmake_minimum_required(VERSION 3.25)
project(IndependentInstalledShvConsumer LANGUAGES CXX)
find_package(SymphonyShvGraphAdapter 0.1.0 EXACT REQUIRED CONFIG)

 message(FATAL_ERROR "unexpected exact adapter package release")

target_link_libraries(consumer PRIVATE Symphony::ShvGraphAdapter)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
