get_filename_component(SYMPHONY_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

project(SymphonyShvGraphAdapter VERSION ${SHV_OWNER_VERSION} LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_CONFIGURATION_TYPES)
 message(FATAL_ERROR "shv-graph-adapter requires a single-configuration generator for exact receipt-owned CMake exports")
endif()
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
string(TOLOWER "${CMAKE_BUILD_TYPE}" CONNECTOR_BUILD_TYPE_LOWER)

include("${SYMPHONY_REPOSITORY_ROOT}/cmake/SymphonyInstallReceiptV2.cmake")
option(SYMPHONY_KVE_USE_INSTALLED "Use installed foundation" OFF)
if(SYMPHONY_KVE_USE_INSTALLED)
 find_package(SymphonyKnowledgeVectorEngine 0.1.0 EXACT REQUIRED CONFIG)
elseif(NOT TARGET Symphony::KnowledgeVectorEngine)
 set(SYMPHONY_KVE_BUILD_TESTING OFF CACHE BOOL "Foundation tests separate" FORCE)
 add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/foundation" EXCLUDE_FROM_ALL)
endif()
set(CONNECTOR_MODULE shv-graph-adapter)
set(CONNECTOR_EXECUTABLE symphony-shv-graph-adapter)
set(CONNECTOR_VERSION ${PROJECT_VERSION}-dev)
set(CONNECTOR_RUNTIME "libexec/symphony/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
set(CONNECTOR_DOCS "share/doc/symphony/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
set(CONNECTOR_SCHEMAS "share/symphony/schemas/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
set(CONNECTOR_INCLUDE_ROOT "${CMAKE_INSTALL_INCLUDEDIR}/symphony/shv-graph-adapter/${CONNECTOR_VERSION}")
set(CONNECTOR_HEADERS "${CONNECTOR_INCLUDE_ROOT}/symphony/graph")
set(CONNECTOR_ARCHIVES "${CMAKE_INSTALL_LIBDIR}/symphony/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
set(CONNECTOR_CMAKE "${CMAKE_INSTALL_LIBDIR}/cmake/SymphonyShvGraphAdapter-${CONNECTOR_VERSION}")
set(CONNECTOR_EXPORT_CONFIG "SymphonyShvGraphAdapterTargets-${CONNECTOR_BUILD_TYPE_LOWER}.cmake")
set(CONNECTOR_LICENSES "share/licenses/symphony-${CONNECTOR_MODULE}/${CONNECTOR_VERSION}")
# Apple ar/ranlib otherwise embed build-time member/symbol-table timestamps.
# Keep the newly public archive reproducible without altering executable code.
if(APPLE)
 set(CMAKE_CXX_ARCHIVE_CREATE "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_AR> qc <TARGET> <LINK_FLAGS> <OBJECTS>")
 set(CMAKE_CXX_ARCHIVE_APPEND "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_AR> q <TARGET> <LINK_FLAGS> <OBJECTS>")
 set(CMAKE_CXX_ARCHIVE_FINISH "\"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 <CMAKE_RANLIB> <TARGET>")
endif()
add_library(shv-graph-adapter-core STATIC src/adapter.cpp)
add_library(Symphony::ShvGraphAdapter ALIAS shv-graph-adapter-core)
set_target_properties(shv-graph-adapter-core PROPERTIES EXPORT_NAME ShvGraphAdapter)
target_include_directories(shv-graph-adapter-core
 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CONNECTOR_INCLUDE_ROOT}>
 PRIVATE src)
target_compile_features(shv-graph-adapter-core PUBLIC cxx_std_26)
target_link_libraries(shv-graph-adapter-core PUBLIC Symphony::KnowledgeVectorEngine)
add_executable(${CONNECTOR_EXECUTABLE} src/main.cpp)
target_link_libraries(${CONNECTOR_EXECUTABLE} PRIVATE shv-graph-adapter-core)
foreach(target shv-graph-adapter-core ${CONNECTOR_EXECUTABLE})
 set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
symphony_install_receipt_v2_preflight(RECEIPT_PATH "share/symphony/receipts/${CONNECTOR_MODULE}/${CONNECTOR_VERSION}/install-receipt.json")
set(CONNECTOR_OWNED "${CONNECTOR_RUNTIME}/${CONNECTOR_EXECUTABLE}|executable")
set(CONNECTOR_INSTALLED "${CONNECTOR_RUNTIME}/${CONNECTOR_EXECUTABLE}")
install(TARGETS ${CONNECTOR_EXECUTABLE} RUNTIME DESTINATION "${CONNECTOR_RUNTIME}")
install(TARGETS shv-graph-adapter-core EXPORT SymphonyShvGraphAdapterTargets
 ARCHIVE DESTINATION "${CONNECTOR_ARCHIVES}")
if(APPLE)
 # CMake invokes ranlib again while installing; reseal its metadata before receipt hashing.
 install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" -E env ZERO_AR_DATE=1 \"${CMAKE_RANLIB}\" \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/${CONNECTOR_ARCHIVES}/${CMAKE_STATIC_LIBRARY_PREFIX}shv-graph-adapter-core${CMAKE_STATIC_LIBRARY_SUFFIX}\" COMMAND_ERROR_IS_FATAL ANY)")
endif()
configure_package_config_file(cmake/SymphonyShvGraphAdapterConfig.cmake.in
 "${CMAKE_CURRENT_BINARY_DIR}/SymphonyShvGraphAdapterConfig.cmake"
 INSTALL_DESTINATION "${CONNECTOR_CMAKE}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/SymphonyShvGraphAdapterConfigVersion.cmake"
 VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(EXPORT SymphonyShvGraphAdapterTargets FILE SymphonyShvGraphAdapterTargets.cmake
 NAMESPACE Symphony:: DESTINATION "${CONNECTOR_CMAKE}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SymphonyShvGraphAdapterConfig.cmake"
 "${CMAKE_CURRENT_BINARY_DIR}/SymphonyShvGraphAdapterConfigVersion.cmake" DESTINATION "${CONNECTOR_CMAKE}")
foreach(sdk_file
 "${CONNECTOR_ARCHIVES}/${CMAKE_STATIC_LIBRARY_PREFIX}shv-graph-adapter-core${CMAKE_STATIC_LIBRARY_SUFFIX}"
 "${CONNECTOR_CMAKE}/SymphonyShvGraphAdapterConfig.cmake"
 "${CONNECTOR_CMAKE}/SymphonyShvGraphAdapterConfigVersion.cmake"
 "${CONNECTOR_CMAKE}/SymphonyShvGraphAdapterTargets.cmake"
 "${CONNECTOR_CMAKE}/${CONNECTOR_EXPORT_CONFIG}")
 list(APPEND CONNECTOR_OWNED "${sdk_file}|regular")
 list(APPEND CONNECTOR_INSTALLED "${sdk_file}")
endforeach()
foreach(doc INTENT.md MANIFEST.md SPEC.md SKILL.md INSTALL.md FEATURES.md)
 list(APPEND CONNECTOR_OWNED "${CONNECTOR_DOCS}/${doc}|regular")
 list(APPEND CONNECTOR_INSTALLED "${CONNECTOR_DOCS}/${doc}")
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" DESTINATION "${CONNECTOR_DOCS}")
endforeach()
foreach(name graph-adapter.schema.json graph-adapter.templates.json)
 list(APPEND CONNECTOR_OWNED "${CONNECTOR_SCHEMAS}/${name}|regular")
 list(APPEND CONNECTOR_INSTALLED "${CONNECTOR_SCHEMAS}/${name}")
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/schemas/v1/${name}" DESTINATION "${CONNECTOR_SCHEMAS}")
endforeach()
list(APPEND CONNECTOR_OWNED "${CONNECTOR_HEADERS}/adapter.hpp|regular" "${CONNECTOR_LICENSES}/LICENSE-AGPL-3.0|regular" "${CONNECTOR_LICENSES}/nlohmann-json-LICENSE.MIT|regular")

install(FILES include/symphony/graph/adapter.hpp DESTINATION "${CONNECTOR_HEADERS}")

install(FILES "${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT" DESTINATION "${CONNECTOR_LICENSES}" RENAME nlohmann-json-LICENSE.MIT)

 list(APPEND CONNECTOR_INSTALLED "${CONNECTOR_DOCS}/${name}")
endforeach()
