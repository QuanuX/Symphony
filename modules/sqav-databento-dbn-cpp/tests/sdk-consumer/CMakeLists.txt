cmake_minimum_required(VERSION 3.30)
project(IndependentInstalledSqavDatabentoDbnConsumer LANGUAGES CXX)
find_package(SymphonySqavDatabentoDbn 0.4.0 EXACT REQUIRED CONFIG)
if(NOT SymphonySqavDatabentoDbn_RELEASE_VERSION STREQUAL "0.4.0-dev")
    message(FATAL_ERROR "unexpected exact SQAV_DBN release")
endif()
add_executable(consumer main.cpp)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(consumer PRIVATE Symphony::SqavDatabentoDbn)
