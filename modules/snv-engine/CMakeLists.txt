cmake_minimum_required(VERSION 3.30)
project(SymphonySnv VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(../../cmake/SymphonySnvEngine.cmake)
option(SYMPHONY_SNV_USE_INSTALLED_DEPENDENCIES "Use exact installed SDKs" OFF)
foreach(owner sniv snrv sciv scnv)
  string(SUBSTRING ${owner} 0 1 first)
  string(TOUPPER ${first} first)
  string(SUBSTRING ${owner} 1 -1 rest)
  set(sdk "${first}${rest}")
  if(SYMPHONY_SNV_USE_INSTALLED_DEPENDENCIES)
    find_package(Symphony${sdk} 0.1.0 EXACT REQUIRED CONFIG)
    symphony_snv_verify_sdk(Symphony::${sdk} ${owner}-engine 0.1.0-dev)
  elseif(NOT TARGET Symphony::${sdk})
    add_subdirectory(../${owner}-engine "${CMAKE_BINARY_DIR}/${owner}")
  endif()
endforeach()
symphony_snv_engine(OWNER snv SDK Snv SOURCES src/snv.cpp TEST_SOURCES tests/snv_test.cpp
  EXTRA_LIBRARIES Symphony::Sniv Symphony::Snrv Symphony::Sciv Symphony::Scnv
  DEPENDENCIES SymphonySniv SymphonySnrv SymphonySciv SymphonyScnv)
if(TARGET snv-engine)
  target_compile_definitions(snv-engine PRIVATE "SNV_EMBEDDED=symphony::snv::snv::embedded_dependencies()")
endif()
