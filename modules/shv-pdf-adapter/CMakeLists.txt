cmake_minimum_required(VERSION 3.25)
project(SymphonyShvPdfAdapter VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
if(NOT CMAKE_INSTALL_LIBEXECDIR STREQUAL "libexec" OR NOT CMAKE_INSTALL_DATADIR STREQUAL "share")
 message(FATAL_ERROR "SHV requires exact libexec/share receipt layout")
endif()
get_filename_component(SYMPHONY_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${SYMPHONY_REPOSITORY_ROOT}/cmake/SymphonyInstallReceiptV2.cmake")
option(SYMPHONY_KVE_USE_INSTALLED "Use independently installed foundation" OFF)
if(SYMPHONY_KVE_USE_INSTALLED)
 find_package(SymphonyKnowledgeVectorEngine 0.1 REQUIRED CONFIG)
elseif(NOT TARGET Symphony::KnowledgeVectorEngine)
 set(SYMPHONY_KVE_BUILD_TESTING OFF CACHE BOOL "Foundation tests run separately" FORCE)
 add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/knowledge-vector-engine-cpp" EXCLUDE_FROM_ALL)
endif()
add_library(shv-pdf-core STATIC src/pdf.cpp src/descriptor.cpp)
target_include_directories(shv-pdf-core PUBLIC src)
target_link_libraries(shv-pdf-core PUBLIC Symphony::KnowledgeVectorEngine ${CMAKE_DL_LIBS})
add_executable(symphony-shv-pdf src/main.cpp)
target_link_libraries(symphony-shv-pdf PRIVATE shv-pdf-core)
foreach(target shv-pdf-core symphony-shv-pdf)
 set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endforeach()
set(SHV_RUNTIME "libexec/symphony/shv-pdf-adapter/0.1.0-dev")
set(SHV_DOCS "share/doc/symphony/shv-pdf-adapter/0.1.0-dev")
set(SHV_SCHEMAS "share/symphony/schemas/shv-pdf-adapter/0.1.0-dev")
set(SHV_LICENSES "share/licenses/symphony-shv-pdf-adapter/0.1.0-dev")
set(SHV_OWNED "${SHV_RUNTIME}/symphony-shv-pdf|executable")
set(SHV_INSTALLED "${SHV_RUNTIME}/symphony-shv-pdf")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "share/symphony/receipts/shv-pdf-adapter/0.1.0-dev/install-receipt.json")
install(TARGETS symphony-shv-pdf RUNTIME DESTINATION "${SHV_RUNTIME}")
foreach(doc INTENT.md MANIFEST.md SPEC.md SKILL.md INSTALL.md FEATURES.md)
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" DESTINATION "${SHV_DOCS}")
 list(APPEND SHV_OWNED "${SHV_DOCS}/${doc}|regular")
 list(APPEND SHV_INSTALLED "${SHV_DOCS}/${doc}")
endforeach()
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/knowledge/shv/PDF-ADAPTER.md" DESTINATION "${SHV_DOCS}")
list(APPEND SHV_OWNED "${SHV_DOCS}/PDF-ADAPTER.md|regular")
list(APPEND SHV_INSTALLED "${SHV_DOCS}/PDF-ADAPTER.md")
foreach(schema pdf.schema.json pdf.templates.json)
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/schemas/v1/${schema}" DESTINATION "${SHV_SCHEMAS}")
 list(APPEND SHV_OWNED "${SHV_SCHEMAS}/${schema}|regular")
 list(APPEND SHV_INSTALLED "${SHV_SCHEMAS}/${schema}")
endforeach()
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/LICENSE" DESTINATION "${SHV_LICENSES}" RENAME LICENSE-AGPL-3.0)
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT" DESTINATION "${SHV_LICENSES}" RENAME nlohmann-json-LICENSE.MIT)
foreach(name LICENSE-AGPL-3.0 nlohmann-json-LICENSE.MIT)
 list(APPEND SHV_OWNED "${SHV_LICENSES}/${name}|regular")
 list(APPEND SHV_INSTALLED "${SHV_LICENSES}/${name}")
endforeach()
symphony_install_receipt_v2(COMPONENT_ID shv-pdf-adapter COMPONENT_KIND adapter MODULE_ID shv-pdf-adapter VECTOR_ID shv ENGINE_ID symphony-shv-pdf PACKAGE_ID shv-pdf-adapter VERSION 0.1.0-dev
 RECEIPT_PATH "share/symphony/receipts/shv-pdf-adapter/0.1.0-dev/install-receipt.json" OWNED_FILES ${SHV_OWNED}
 ENTRY_POINTS "symphony-shv-pdf|executable|${SHV_RUNTIME}/symphony-shv-pdf|symphony.knowledge.engine-process.v1")
configure_file(cmake/uninstall.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" @ONLY)
add_custom_target(uninstall-shv-pdf-adapter COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")

if(BUILD_TESTING)
 add_executable(shv-pdf-test tests/pdf_test.cpp)
 target_link_libraries(shv-pdf-test PRIVATE shv-pdf-core)
 set_target_properties(shv-pdf-test PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 add_test(NAME shv-pdf-bounded-table COMMAND shv-pdf-test)
endif()
