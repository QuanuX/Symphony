 find_package(SymphonyKnowledgeVectorEngine 0.1 REQUIRED CONFIG)

 add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/knowledge-vector-engine-cpp" EXCLUDE_FROM_ALL)
endif()

target_include_directories(shv-document-reader PUBLIC ../shv-pdf-adapter/src)
target_link_libraries(shv-document-reader PUBLIC Symphony::KnowledgeVectorEngine ${CMAKE_DL_LIBS})
add_library(shv-core STATIC src/common.cpp src/coverage.cpp src/html.cpp src/table.cpp src/catalogue.cpp src/evaluation.cpp src/descriptor.cpp)
target_include_directories(shv-core PUBLIC src)
target_link_libraries(shv-core PUBLIC Symphony::KnowledgeVectorEngine shv-document-reader)
add_executable(symphony-shv src/main.cpp)
target_link_libraries(symphony-shv PRIVATE shv-core)
foreach(target shv-core symphony-shv shv-document-reader)
 set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endforeach()

 list(APPEND SHV_OWNED "${SHV_DOCS}/${doc}|regular")
 list(APPEND SHV_INSTALLED "${SHV_DOCS}/${doc}")
endforeach()
foreach(schema_path IN LISTS SHV_OWNER_SCHEMA_FILES)
 get_filename_component(schema "${schema_path}" NAME)
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/schemas/v1/${schema}" DESTINATION "${SHV_SCHEMAS}")
 list(APPEND SHV_OWNED "${SHV_SCHEMAS}/${schema}|regular")
 list(APPEND SHV_INSTALLED "${SHV_SCHEMAS}/${schema}")
endforeach()

endforeach()
