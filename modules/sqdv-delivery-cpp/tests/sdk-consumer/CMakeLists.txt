cmake_minimum_required(VERSION 3.30)
project(IndependentInstalledSqdvDeliveryConsumer LANGUAGES CXX)
find_package(SymphonySqdvDelivery 0.1.0 EXACT REQUIRED CONFIG)
if(NOT SymphonySqdvDelivery_RELEASE_VERSION STREQUAL "0.1.0-dev")
    message(FATAL_ERROR "unexpected exact SQDV release")
endif()
add_executable(consumer main.cpp)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(consumer PRIVATE Symphony::SqdvDelivery)
