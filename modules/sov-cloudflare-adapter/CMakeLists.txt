cmake_minimum_required(VERSION 3.30)
project(SymphonySovCloudflareAdapter VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ROOT}/cmake/SymphonyInstallReceiptV2.cmake")
option(SYMPHONY_CLOUDFLARE_INSTALLED "Use the exact installed foundation SDK without source fallback" OFF)
if(SYMPHONY_CLOUDFLARE_INSTALLED)
  find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT REQUIRED CONFIG)
elseif(NOT TARGET Symphony::KnowledgeVectorEngine)
  function(cloudflare_foundation)
    set(BUILD_TESTING OFF)
    set(SYMPHONY_KVE_BUILD_TESTING OFF)
    add_subdirectory("${ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/foundation" EXCLUDE_FROM_ALL)
  endfunction()
  cloudflare_foundation()
endif()
if(SYMPHONY_CLOUDFLARE_INSTALLED)
  find_package(SymphonySSIAGPrivateStore 0.1.0 EXACT REQUIRED CONFIG)
elseif(NOT TARGET Symphony::SSIAGPrivateChannel)
  add_subdirectory("${ROOT}/modules/ssiag-provider-private-store" "${CMAKE_CURRENT_BINARY_DIR}/credential-provider" EXCLUDE_FROM_ALL)
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${ROOT}/knowledge/sov/cloudflare/owner.json"
  "${CMAKE_CURRENT_SOURCE_DIR}/runtime/request-guard.mjs")
file(READ "${ROOT}/knowledge/sov/cloudflare/owner.json" CLOUDFLARE_OWNER_JSON)
string(JSON CLOUDFLARE_OWNER_JSON GET "${CLOUDFLARE_OWNER_JSON}")
# CMake's JSON formatter retains indentation. Split long literal storage into
# an array of byte values instead of relying on compiler overlength extensions.
file(READ "${ROOT}/knowledge/sov/cloudflare/owner.json" OWNER_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," OWNER_BYTES "${OWNER_HEX}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/owner_data.hpp" "#pragma once\ninline constexpr unsigned char owner_json[] = {${OWNER_BYTES}0};\n")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/runtime/request-guard.mjs" GUARD_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," GUARD_BYTES "${GUARD_HEX}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/request_guard.hpp" "#pragma once\ninline constexpr unsigned char request_guard[] = {${GUARD_BYTES}0};\n")
add_library(sov-cloudflare-core STATIC src/adapter.cpp src/resource.cpp src/credential_channel.cpp src/http_runner.cpp src/execution.cpp src/connector.cpp)
target_include_directories(sov-cloudflare-core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sov-cloudflare-core PUBLIC Symphony::KnowledgeVectorEngine Symphony::SSIAGPrivateChannel)
add_executable(symphony-sov-cloudflare-adapter src/main.cpp)
target_link_libraries(symphony-sov-cloudflare-adapter PRIVATE sov-cloudflare-core)
foreach(tool generate prepare_bundle)
  add_executable(cloudflare-${tool} "tools/${tool}.cpp")
  target_link_libraries(cloudflare-${tool} PRIVATE sov-cloudflare-core)
  set_target_properties(cloudflare-${tool} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
endforeach()
foreach(t sov-cloudflare-core symphony-sov-cloudflare-adapter)
  set_target_properties(${t} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_options(${t} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endforeach()
set(VERSION_FULL "0.1.0-dev")
set(RUNTIME "libexec/symphony/sov-cloudflare-adapter/${VERSION_FULL}")
set(DOCS "share/doc/symphony/sov-cloudflare-adapter/${VERSION_FULL}")
set(SCHEMAS "share/symphony/schemas/sov-cloudflare-adapter/${VERSION_FULL}")
set(RECEIPT "share/symphony/receipts/sov-cloudflare-adapter/${VERSION_FULL}/install-receipt.json")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "${RECEIPT}")
set(OWNED "${RUNTIME}/symphony-sov-cloudflare-adapter|executable")
install(TARGETS symphony-sov-cloudflare-adapter RUNTIME DESTINATION "${RUNTIME}")
foreach(doc INTENT.md MANIFEST.md SPEC.md INSTALL.md SKILL.md FEATURES.md)
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" DESTINATION "${DOCS}")
  list(APPEND OWNED "${DOCS}/${doc}|regular")
endforeach()
install(FILES "${ROOT}/knowledge/sov/cloudflare/owner.json" DESTINATION "${SCHEMAS}")
list(APPEND OWNED "${SCHEMAS}/owner.json|regular")
file(GLOB companions CONFIGURE_DEPENDS "${ROOT}/knowledge/sov/cloudflare/schemas/*.json")
foreach(path IN LISTS companions)
  get_filename_component(name "${path}" NAME)
  install(FILES "${path}" DESTINATION "${SCHEMAS}")
  list(APPEND OWNED "${SCHEMAS}/${name}|regular")
endforeach()
install(FILES "${ROOT}/LICENSE" DESTINATION "${DOCS}" RENAME LICENSE-AGPL-3.0)
install(FILES "${ROOT}/libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT" DESTINATION "${DOCS}" RENAME nlohmann-json-LICENSE.MIT)
list(APPEND OWNED "${DOCS}/LICENSE-AGPL-3.0|regular" "${DOCS}/nlohmann-json-LICENSE.MIT|regular")
symphony_install_receipt_v2(COMPONENT_ID sov-cloudflare-adapter COMPONENT_KIND adapter
  MODULE_ID sov-cloudflare-adapter VECTOR_ID sov ENGINE_ID symphony-sov-cloudflare-adapter
  PACKAGE_ID sov-cloudflare-adapter VERSION "${VERSION_FULL}" RECEIPT_PATH "${RECEIPT}"
  OWNED_FILES ${OWNED}
  ENTRY_POINTS "symphony-sov-cloudflare-adapter|executable|${RUNTIME}/symphony-sov-cloudflare-adapter|symphony.sov.cloudflare-request.v1")
set(INSTALLED)
foreach(item IN LISTS OWNED)
  string(REGEX REPLACE "[|].*$" "" path "${item}")
  list(APPEND INSTALLED "${path}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake"
  "if(NOT DEFINED INSTALL_PREFIX)\n set(INSTALL_PREFIX \"${CMAKE_INSTALL_PREFIX}\")\nendif()\nset(_files ${INSTALLED} \"${RECEIPT}\")\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/SymphonyUninstallReceiptV2.cmake\")\n")
add_custom_target(uninstall-sov-cloudflare-adapter COMMAND ${CMAKE_COMMAND}
  -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
if(BUILD_TESTING)
  add_test(NAME cloudflare-owner-projections COMMAND cloudflare-generate "${ROOT}" --check)
  foreach(test contract qualification execution connector http_runner)
    add_executable(cloudflare-${test}-tests "tests/${test}.cpp")
    target_include_directories(cloudflare-${test}-tests PRIVATE "${ROOT}/libraries/knowledge-vector-engine-cpp/tests/support")
    target_link_libraries(cloudflare-${test}-tests PRIVATE sov-cloudflare-core)
    set_target_properties(cloudflare-${test}-tests PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  endforeach()
  add_test(NAME cloudflare-native-contract COMMAND cloudflare-contract-tests $<TARGET_FILE:symphony-sov-cloudflare-adapter>)
  add_executable(cloudflare-connector-child tests/connector_child.cpp)
  set_target_properties(cloudflare-connector-child PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON)
  if(TARGET symphony-ssiag-provider-private-store)
    set(CLOUDFLARE_PRIVATE_PROVIDER $<TARGET_FILE:symphony-ssiag-provider-private-store>)
    add_dependencies(cloudflare-execution-tests symphony-ssiag-provider-private-store ssiag-private-contract-tests)
    add_dependencies(cloudflare-connector-tests symphony-ssiag-provider-private-store)
  else()
    find_program(CLOUDFLARE_PRIVATE_PROVIDER NAMES symphony-ssiag-provider-private-store HINTS ${CMAKE_PREFIX_PATH} PATH_SUFFIXES libexec/symphony/ssiag-provider-private-store/0.1.0-dev NO_DEFAULT_PATH REQUIRED)
  endif()
  add_test(NAME cloudflare-resource-execution COMMAND cloudflare-execution-tests ${CLOUDFLARE_PRIVATE_PROVIDER})
  add_test(NAME cloudflare-connector-lifecycle COMMAND cloudflare-connector-tests ${CLOUDFLARE_PRIVATE_PROVIDER} $<TARGET_FILE:cloudflare-connector-child>)
  set_tests_properties(cloudflare-connector-lifecycle PROPERTIES ENVIRONMENT "TMPDIR=/tmp" TIMEOUT 90)
  add_executable(cloudflare-installed-tests tests/installed.cpp)
  target_include_directories(cloudflare-installed-tests PRIVATE "${ROOT}/libraries/knowledge-vector-engine-cpp/tests/support")
  target_link_libraries(cloudflare-installed-tests PRIVATE sov-cloudflare-core)
  set_target_properties(cloudflare-installed-tests PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
