cmake_minimum_required(VERSION 3.30)
project(SymphonySnvLocalObserver VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/SymphonySnvEngine.cmake")
if(CMAKE_CONFIGURATION_TYPES)
  message(FATAL_ERROR "Observer receipt requires a single configuration generator")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(SYMPHONY_SNV_USE_INSTALLED_DEPENDENCIES "Use exact receipt-bound installed SDKs" OFF)
if(SYMPHONY_SNV_USE_INSTALLED_DEPENDENCIES)
  find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT REQUIRED CONFIG)
  symphony_snv_verify_sdk(Symphony::KnowledgeVectorEngine knowledge-vector-engine-cpp 0.2.0-dev)
else()
  if(NOT TARGET Symphony::KnowledgeVectorEngine)
    set(SYMPHONY_KVE_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("../../libraries/knowledge-vector-engine-cpp" "${CMAKE_BINARY_DIR}/foundation")
  endif()
endif()
set(release "0.1.0-dev")
set(component "snv-local-observer")
set(inc "${CMAKE_INSTALL_INCLUDEDIR}/symphony/${component}/${release}")
set(lib "${CMAKE_INSTALL_LIBDIR}/symphony/${component}/${release}")
set(config "${CMAKE_INSTALL_LIBDIR}/cmake/SymphonySnvLocalObserver-${release}")
set(share "share/symphony/${component}/${release}")
set(receipt "share/symphony/receipts/${component}/${release}/install-receipt.json")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "${receipt}")
add_library(symphony-snv-local-observer STATIC src/observer.cpp)
add_library(Symphony::SnvLocalObserver ALIAS symphony-snv-local-observer)
set_target_properties(symphony-snv-local-observer PROPERTIES EXPORT_NAME SnvLocalObserver)
target_include_directories(symphony-snv-local-observer PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${inc}>)
target_link_libraries(symphony-snv-local-observer PUBLIC Symphony::KnowledgeVectorEngine)
add_executable(snv-local-observer src/main.cpp)
set_target_properties(snv-local-observer PROPERTIES OUTPUT_NAME symphony-snv-local-observer)
target_link_libraries(snv-local-observer PRIVATE symphony-snv-local-observer)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/interface.generated.hpp")
  target_compile_definitions(snv-local-observer PRIVATE OBSERVER_GENERATED_INTERFACE=1)
endif()
foreach(target symphony-snv-local-observer snv-local-observer)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_features(${target} PUBLIC cxx_std_26)
  target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endforeach()
install(TARGETS snv-local-observer RUNTIME DESTINATION "libexec/symphony/${component}/${release}")
install(TARGETS symphony-snv-local-observer EXPORT SymphonySnvLocalObserverTargets ARCHIVE DESTINATION "${lib}")
install(DIRECTORY include/ DESTINATION "${inc}")
set(owned "libexec/symphony/${component}/${release}/symphony-snv-local-observer|executable"
  "${lib}/${CMAKE_STATIC_LIBRARY_PREFIX}symphony-snv-local-observer${CMAKE_STATIC_LIBRARY_SUFFIX}|regular"
  "${inc}/symphony/snv/local_observer.hpp|regular")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/SymphonySnvLocalObserverConfig.cmake" "include(CMakeFindDependencyMacro)\nfind_dependency(SymphonyKnowledgeVectorEngine 0.2.0 EXACT CONFIG)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/SymphonySnvLocalObserverTargets.cmake\")\n")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/SymphonySnvLocalObserverConfigVersion.cmake" VERSION 0.1.0 COMPATIBILITY ExactVersion)
install(EXPORT SymphonySnvLocalObserverTargets FILE SymphonySnvLocalObserverTargets.cmake NAMESPACE Symphony:: DESTINATION "${config}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SymphonySnvLocalObserverConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/SymphonySnvLocalObserverConfigVersion.cmake" DESTINATION "${config}")
string(TOLOWER "${CMAKE_BUILD_TYPE}" build_type)
foreach(f Config.cmake ConfigVersion.cmake Targets.cmake Targets-${build_type}.cmake)
  list(APPEND owned "${config}/SymphonySnvLocalObserver${f}|regular")
endforeach()
foreach(doc INTENT.md MANIFEST.md INSTALL.md SKILL.md SPEC.md FEATURES.md)
  install(FILES "${doc}" DESTINATION "share/doc/symphony/${component}/${release}")
  list(APPEND owned "share/doc/symphony/${component}/${release}/${doc}|regular")
endforeach()
foreach(meta OWNER-INTERFACE.json INTERFACE-GENERATOR.json)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${meta}")
    install(FILES "${meta}" DESTINATION "${share}")
    list(APPEND owned "${share}/${meta}|regular")
  endif()
endforeach()
file(GLOB resources RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/schemas/v1/*.json")
foreach(resource IN LISTS resources)
  get_filename_component(name "${resource}" NAME)
  install(FILES "${resource}" DESTINATION "${share}/schemas/v1")
  list(APPEND owned "${share}/schemas/v1/${name}|regular")
endforeach()
install(FILES "../../LICENSE" DESTINATION "share/licenses/symphony-${component}/${release}" RENAME LICENSE-AGPL-3.0)
install(FILES "../../libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT" DESTINATION "share/licenses/symphony-${component}/${release}" RENAME nlohmann-json-LICENSE.MIT)
list(APPEND owned "share/licenses/symphony-${component}/${release}/LICENSE-AGPL-3.0|regular" "share/licenses/symphony-${component}/${release}/nlohmann-json-LICENSE.MIT|regular")
symphony_install_receipt_v2(COMPONENT_ID "${component}" COMPONENT_KIND module MODULE_ID "${component}" PACKAGE_ID "${component}" VERSION "${release}" RECEIPT_PATH "${receipt}" OWNED_FILES ${owned} ENTRY_POINTS "symphony-snv-local-observer|executable|libexec/symphony/${component}/${release}/symphony-snv-local-observer|symphony.knowledge.engine-process.v1")
set(uninstall_files "")
foreach(spec IN LISTS owned)
  string(REPLACE "|" ";" fields "${spec}")
  list(GET fields 0 path)
  string(APPEND uninstall_files " [==[${path}]==]")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" "if(NOT DEFINED INSTALL_PREFIX)\nset(INSTALL_PREFIX [==[${CMAKE_INSTALL_PREFIX}]==])\nendif()\nset(_files${uninstall_files} [==[${receipt}]==])\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/SymphonyUninstallReceiptV2.cmake\")\n")
add_custom_target(uninstall-snv-local-observer COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
if(BUILD_TESTING)
  add_executable(snv-local-observer-tests tests/observer_test.cpp)
  target_link_libraries(snv-local-observer-tests PRIVATE symphony-snv-local-observer)
  set_target_properties(snv-local-observer-tests PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_options(snv-local-observer-tests PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
  add_test(NAME snv-local-observer-tests COMMAND snv-local-observer-tests)
  add_executable(snv-local-observer-process-tests tests/process.cpp)
  target_link_libraries(snv-local-observer-process-tests PRIVATE symphony-snv-local-observer)
  target_include_directories(snv-local-observer-process-tests PRIVATE "../../libraries/knowledge-vector-engine-cpp/tests/support")
  set_target_properties(snv-local-observer-process-tests PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_options(snv-local-observer-process-tests PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endif()
