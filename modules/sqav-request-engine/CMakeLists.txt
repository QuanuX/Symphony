  find_package(SymphonySqavDatabentoReference 0.1.0 EXACT REQUIRED CONFIG)
else()
  function(sqav_request_dependencies)
    set(BUILD_TESTING OFF)
    if(NOT TARGET Symphony::SqavFred)
      add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/modules/sqav-fred-cpp" "${CMAKE_CURRENT_BINARY_DIR}/fred" EXCLUDE_FROM_ALL)
    endif()
    if(NOT TARGET Symphony::SqavDatabentoReference)
      add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/modules/sqav-databento-reference-cpp" "${CMAKE_CURRENT_BINARY_DIR}/reference" EXCLUDE_FROM_ALL)
    endif()
  endfunction()
  sqav_request_dependencies()
  foreach(binding "sqav-fred-cpp|SymphonySqavFred|0.1.0" "sqav-databento-dbn-cpp|SymphonySqavDatabentoDbn|0.5.0" "sqav-databento-reference-cpp|SymphonySqavDatabentoReference|0.1.0")
    string(REPLACE "|" ";" parts "${binding}")
    list(GET parts 0 module)
    list(GET parts 1 project_name)
    list(GET parts 2 required_version)
    get_directory_property(source_version DIRECTORY "${SYMPHONY_REPOSITORY_ROOT}/modules/${module}" DEFINITION ${project_name}_VERSION)
    if(NOT source_version STREQUAL required_version)
      message(FATAL_ERROR "SQAV request requires ${module} ${required_version}; source version differs")
    endif()
  endforeach()
endif()

  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
