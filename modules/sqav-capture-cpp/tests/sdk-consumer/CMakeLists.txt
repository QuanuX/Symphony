cmake_minimum_required(VERSION 3.30)
project(IndependentInstalledSqavCaptureConsumer LANGUAGES CXX)
find_package(SymphonySqavCapture 0.1.0 EXACT REQUIRED CONFIG)
if(NOT SymphonySqavCapture_RELEASE_VERSION STREQUAL "0.1.0-dev")
    message(FATAL_ERROR "unexpected exact SQAV release")
endif()
add_executable(consumer main.cpp)
set_target_properties(consumer PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(consumer PRIVATE Symphony::SqavCapture)
