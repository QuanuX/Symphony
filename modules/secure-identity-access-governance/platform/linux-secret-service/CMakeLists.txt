cmake_minimum_required(VERSION 3.30)
project(SymphonySSIAGLinuxSecretService VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_CONFIGURATION_TYPES)
  message(FATAL_ERROR "Linux and a single-configuration generator are required")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
include(CTest)
include(CMakePackageConfigHelpers)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../../../cmake/SymphonyInstallReceiptV2.cmake")
find_package(PkgConfig REQUIRED)
pkg_check_modules(LibSecret REQUIRED IMPORTED_TARGET libsecret-1>=0.21.2)
set(component ssiag-linux-secret-service)
set(release 0.1.0-dev)
set(inc "include/symphony/${component}/${release}")
set(lib "lib/symphony/${component}/${release}")
set(config "lib/cmake/${PROJECT_NAME}-${release}")
set(receipt "share/symphony/receipts/${component}/${release}/install-receipt.json")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "${receipt}")
add_library(${component} STATIC src/provider.cpp)
add_library(Symphony::SSIAGLinuxSecretService ALIAS ${component})
set_target_properties(${component} PROPERTIES EXPORT_NAME SSIAGLinuxSecretService CXX_EXTENSIONS OFF)
target_compile_features(${component} PUBLIC cxx_std_26)
target_compile_options(${component} PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(${component} PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${inc}>")
target_link_libraries(${component} PUBLIC PkgConfig::LibSecret)
install(TARGETS ${component} EXPORT ${PROJECT_NAME}Targets ARCHIVE DESTINATION "${lib}")
install(DIRECTORY include/ DESTINATION "${inc}")
install(EXPORT ${PROJECT_NAME}Targets NAMESPACE Symphony:: DESTINATION "${config}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}Config.cmake" "include(CMakeFindDependencyMacro)\nfind_dependency(PkgConfig)\npkg_check_modules(LibSecret REQUIRED IMPORTED_TARGET libsecret-1>=0.21.2)\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/${PROJECT_NAME}Targets.cmake\")\n")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}ConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}Config.cmake" "${CMAKE_CURRENT_BINARY_DIR}/${PROJECT_NAME}ConfigVersion.cmake" DESTINATION "${config}")
set(owned "${lib}/lib${component}.a|regular" "${inc}/symphony/ssiag/linux_secret_service.hpp|regular")
string(TOLOWER "${CMAKE_BUILD_TYPE}" build_type)
foreach(file Config.cmake ConfigVersion.cmake Targets.cmake Targets-${build_type}.cmake)
  list(APPEND owned "${config}/${PROJECT_NAME}${file}|regular")
endforeach()
set(license "share/licenses/symphony-${component}/${release}/LICENSE-AGPL-3.0")
install(FILES "../../../../LICENSE" DESTINATION "share/licenses/symphony-${component}/${release}" RENAME LICENSE-AGPL-3.0)
list(APPEND owned "${license}|regular")
symphony_install_receipt_v2(COMPONENT_ID "${component}" COMPONENT_KIND module MODULE_ID "${component}" VECTOR_ID ssiag PACKAGE_ID "${component}" VERSION "${release}" RECEIPT_PATH "${receipt}" OWNED_FILES ${owned})
set(uninstall_files "")
foreach(spec IN LISTS owned)
  string(REPLACE "|" ";" fields "${spec}")
  list(GET fields 0 path)
  string(APPEND uninstall_files " [==[${path}]==]")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" "if(NOT DEFINED INSTALL_PREFIX)\nset(INSTALL_PREFIX [==[${CMAKE_INSTALL_PREFIX}]==])\nendif()\nset(_files${uninstall_files} [==[${receipt}]==])\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/SymphonyUninstallReceiptV2.cmake\")\n")
add_custom_target(uninstall-ssiag-linux-secret-service COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
if(BUILD_TESTING)
  add_executable(linux-secret-service-test tests/provider_test.cpp)
  target_link_libraries(linux-secret-service-test PRIVATE Symphony::SSIAGLinuxSecretService)
  add_test(NAME linux-secret-service-disposable-session COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/private-session.sh" "$<TARGET_FILE:linux-secret-service-test>")
  set_tests_properties(linux-secret-service-disposable-session PROPERTIES TIMEOUT 40)
endif()
