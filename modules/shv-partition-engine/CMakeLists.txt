cmake_minimum_required(VERSION 3.25)
project(SymphonyShvPartitionEngine VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
if(NOT CMAKE_INSTALL_LIBEXECDIR STREQUAL "libexec" OR NOT CMAKE_INSTALL_DATADIR STREQUAL "share")
 message(FATAL_ERROR "SHV requires exact libexec/share receipt layout")
endif()
get_filename_component(SYMPHONY_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${SYMPHONY_REPOSITORY_ROOT}/cmake/SymphonyInstallReceiptV2.cmake")
option(SYMPHONY_KVE_USE_INSTALLED "Use independently installed foundation" OFF)
if(SYMPHONY_KVE_USE_INSTALLED)
 find_package(SymphonyKnowledgeVectorEngine 0.1 REQUIRED CONFIG)
elseif(NOT TARGET Symphony::KnowledgeVectorEngine)
 set(SYMPHONY_KVE_BUILD_TESTING OFF CACHE BOOL "Foundation tests run separately" FORCE)
 add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/knowledge-vector-engine-cpp" EXCLUDE_FROM_ALL)
endif()
add_library(shv-partition-core STATIC src/partition.cpp src/descriptor.cpp)
target_include_directories(shv-partition-core PUBLIC src)
target_link_libraries(shv-partition-core PUBLIC Symphony::KnowledgeVectorEngine)
add_executable(symphony-shv-partition src/main.cpp)
target_link_libraries(symphony-shv-partition PRIVATE shv-partition-core)
foreach(target shv-partition-core symphony-shv-partition)
 set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endforeach()
if(BUILD_TESTING)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME shv-partition-process COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/partition_test.py" --engine $<TARGET_FILE:symphony-shv-partition>)
endif()
set(SHV_RUNTIME "libexec/symphony/shv-partition-engine/0.1.0-dev")
set(SHV_DOCS "share/doc/symphony/shv-partition-engine/0.1.0-dev")
set(SHV_SCHEMAS "share/symphony/schemas/shv-partition-engine/0.1.0-dev")
set(SHV_LICENSES "share/licenses/symphony-shv-partition-engine/0.1.0-dev")
set(SHV_OWNED "${SHV_RUNTIME}/symphony-shv-partition|executable")
set(SHV_INSTALLED "${SHV_RUNTIME}/symphony-shv-partition")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "share/symphony/receipts/shv-partition-engine/0.1.0-dev/install-receipt.json")
install(TARGETS symphony-shv-partition RUNTIME DESTINATION "${SHV_RUNTIME}")
foreach(doc INTENT.md MANIFEST.md SPEC.md SKILL.md INSTALL.md FEATURES.md)
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/${doc}" DESTINATION "${SHV_DOCS}")
 list(APPEND SHV_OWNED "${SHV_DOCS}/${doc}|regular")
 list(APPEND SHV_INSTALLED "${SHV_DOCS}/${doc}")
endforeach()
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/knowledge/shv/PARTITIONS.md" DESTINATION "${SHV_DOCS}")
list(APPEND SHV_OWNED "${SHV_DOCS}/PARTITIONS.md|regular")
list(APPEND SHV_INSTALLED "${SHV_DOCS}/PARTITIONS.md")
foreach(schema partition.schema.json partition.templates.json)
 install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/schemas/v1/${schema}" DESTINATION "${SHV_SCHEMAS}")
 list(APPEND SHV_OWNED "${SHV_SCHEMAS}/${schema}|regular")
 list(APPEND SHV_INSTALLED "${SHV_SCHEMAS}/${schema}")
endforeach()
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/LICENSE" DESTINATION "${SHV_LICENSES}" RENAME LICENSE-AGPL-3.0)
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT" DESTINATION "${SHV_LICENSES}" RENAME nlohmann-json-LICENSE.MIT)
foreach(name LICENSE-AGPL-3.0 nlohmann-json-LICENSE.MIT)
 list(APPEND SHV_OWNED "${SHV_LICENSES}/${name}|regular")
 list(APPEND SHV_INSTALLED "${SHV_LICENSES}/${name}")
endforeach()
symphony_install_receipt_v2(COMPONENT_ID shv-partition-engine COMPONENT_KIND vector_engine MODULE_ID shv-partition-engine VECTOR_ID shv ENGINE_ID symphony-shv-partition PACKAGE_ID shv-partition-engine VERSION 0.1.0-dev
 RECEIPT_PATH "share/symphony/receipts/shv-partition-engine/0.1.0-dev/install-receipt.json" OWNED_FILES ${SHV_OWNED}
 ENTRY_POINTS "symphony-shv-partition|executable|${SHV_RUNTIME}/symphony-shv-partition|symphony.knowledge.engine-process.v1")
configure_file(cmake/uninstall.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" @ONLY)
add_custom_target(uninstall-shv-partition-engine COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
