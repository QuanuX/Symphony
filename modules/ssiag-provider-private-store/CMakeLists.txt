cmake_minimum_required(VERSION 3.30)
project(SymphonySSIAGPrivateStore VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(CMakePackageConfigHelpers)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${ROOT}/cmake/SymphonyInstallReceiptV2.cmake")
option(SYMPHONY_SSIAG_PRIVATE_INSTALLED "Use exact installed foundation" OFF)
if(SYMPHONY_SSIAG_PRIVATE_INSTALLED)
 find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT REQUIRED CONFIG)
elseif(NOT TARGET Symphony::KnowledgeVectorEngine)
 function(private_foundation)
  set(BUILD_TESTING OFF)
  set(SYMPHONY_KVE_BUILD_TESTING OFF)
  add_subdirectory("${ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/foundation" EXCLUDE_FROM_ALL)
 endfunction()
 private_foundation()
endif()
add_library(ssiag-private-channel INTERFACE)
add_library(Symphony::SSIAGPrivateChannel ALIAS ssiag-private-channel)
set_target_properties(ssiag-private-channel PROPERTIES EXPORT_NAME SSIAGPrivateChannel)
target_include_directories(ssiag-private-channel INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include/symphony/ssiag-private-store/0.1.0-dev>")
target_link_libraries(ssiag-private-channel INTERFACE Symphony::KnowledgeVectorEngine)
target_compile_features(ssiag-private-channel INTERFACE cxx_std_26)
add_executable(symphony-ssiag-provider-private-store src/main.cpp)
target_link_libraries(symphony-ssiag-provider-private-store PRIVATE Symphony::SSIAGPrivateChannel)
set_target_properties(symphony-ssiag-provider-private-store PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_options(symphony-ssiag-provider-private-store PRIVATE -Wall -Wextra -Wpedantic -Werror)
set(VERSION_FULL "0.1.0-dev")
set(RUNTIME "libexec/symphony/ssiag-provider-private-store/${VERSION_FULL}")
set(DOCS "share/doc/symphony/ssiag-provider-private-store/${VERSION_FULL}")
set(SDK "lib/cmake/SymphonySSIAGPrivateStore-${VERSION_FULL}")
set(RECEIPT "share/symphony/receipts/ssiag-provider-private-store/${VERSION_FULL}/install-receipt.json")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "${RECEIPT}")
install(TARGETS symphony-ssiag-provider-private-store RUNTIME DESTINATION "${RUNTIME}")
install(TARGETS ssiag-private-channel EXPORT SymphonySSIAGPrivateStoreTargets)
install(EXPORT SymphonySSIAGPrivateStoreTargets FILE SymphonySSIAGPrivateStoreTargets.cmake NAMESPACE Symphony:: DESTINATION "${SDK}")
configure_package_config_file(cmake/SymphonySSIAGPrivateStoreConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/SymphonySSIAGPrivateStoreConfig.cmake" INSTALL_DESTINATION "${SDK}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/SymphonySSIAGPrivateStoreConfigVersion.cmake" VERSION 0.1.0 COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SymphonySSIAGPrivateStoreConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/SymphonySSIAGPrivateStoreConfigVersion.cmake" DESTINATION "${SDK}")
install(FILES include/symphony/ssiag/private_channel.hpp DESTINATION "include/symphony/ssiag-private-store/${VERSION_FULL}/symphony/ssiag")
set(OWNED "${RUNTIME}/symphony-ssiag-provider-private-store|executable" "include/symphony/ssiag-private-store/${VERSION_FULL}/symphony/ssiag/private_channel.hpp|regular" "${SDK}/SymphonySSIAGPrivateStoreConfig.cmake|regular" "${SDK}/SymphonySSIAGPrivateStoreConfigVersion.cmake|regular" "${SDK}/SymphonySSIAGPrivateStoreTargets.cmake|regular")
foreach(doc INTENT.md MANIFEST.md SPEC.md SKILL.md INSTALL.md FEATURES.md)
 install(FILES "${doc}" DESTINATION "${DOCS}")
 list(APPEND OWNED "${DOCS}/${doc}|regular")
endforeach()
install(FILES "${ROOT}/LICENSE" DESTINATION "${DOCS}" RENAME LICENSE-AGPL-3.0)
install(FILES "${ROOT}/libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT" DESTINATION "${DOCS}" RENAME nlohmann-json-LICENSE.MIT)
list(APPEND OWNED "${DOCS}/LICENSE-AGPL-3.0|regular" "${DOCS}/nlohmann-json-LICENSE.MIT|regular")
symphony_install_receipt_v2(COMPONENT_ID ssiag-provider-private-store COMPONENT_KIND adapter MODULE_ID ssiag-provider-private-store VECTOR_ID ssiag ENGINE_ID symphony-ssiag-provider-private-store PACKAGE_ID ssiag-provider-private-store VERSION "${VERSION_FULL}" RECEIPT_PATH "${RECEIPT}" OWNED_FILES ${OWNED} ENTRY_POINTS "symphony-ssiag-provider-private-store|executable|${RUNTIME}/symphony-ssiag-provider-private-store|symphony.ssiag.cloudflare-use.v1")
set(INSTALLED)
foreach(item IN LISTS OWNED)
 string(REGEX REPLACE "[|].*$" "" path "${item}")
 list(APPEND INSTALLED "${path}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" "if(NOT DEFINED INSTALL_PREFIX)\n set(INSTALL_PREFIX \"${CMAKE_INSTALL_PREFIX}\")\nendif()\nset(_files ${INSTALLED} \"${RECEIPT}\")\ninclude(\"\${CMAKE_CURRENT_LIST_DIR}/SymphonyUninstallReceiptV2.cmake\")\n")
add_custom_target(uninstall-ssiag-provider-private-store COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
if(BUILD_TESTING)
 add_executable(ssiag-private-contract-tests tests/contract.cpp)
 target_link_libraries(ssiag-private-contract-tests PRIVATE Symphony::SSIAGPrivateChannel)
 target_include_directories(ssiag-private-contract-tests PRIVATE "${ROOT}/libraries/knowledge-vector-engine-cpp/tests/support")
 set_target_properties(ssiag-private-contract-tests PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 add_test(NAME ssiag-private-store-contract COMMAND ssiag-private-contract-tests $<TARGET_FILE:symphony-ssiag-provider-private-store>)
endif()
