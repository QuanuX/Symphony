if(SYMPHONY_KVE_USE_INSTALLED)
 find_package(SymphonyKnowledgeVectorEngine 0.1.0 EXACT REQUIRED CONFIG)
elseif(NOT TARGET Symphony::KnowledgeVectorEngine)
 set(SYMPHONY_KVE_BUILD_TESTING OFF CACHE BOOL "Foundation tests run separately" FORCE)
 add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/knowledge-vector-engine-cpp" EXCLUDE_FROM_ALL)
endif()
add_library(shv-profile-document STATIC ../shv-pdf-adapter/src/pdf.cpp ../shv-pdf-adapter/src/descriptor.cpp ../shv-pdf-adapter/src/graph.cpp)
target_include_directories(shv-profile-document PUBLIC ../shv-pdf-adapter/src)
target_link_libraries(shv-profile-document PUBLIC Symphony::KnowledgeVectorEngine ${CMAKE_DL_LIBS})
add_library(shv-profile-core STATIC src/profile.cpp src/diagnostics.cpp src/descriptor.cpp
 ../shv-engine/src/common.cpp ../shv-engine/src/coverage.cpp ../shv-engine/src/html.cpp ../shv-engine/src/table.cpp ../shv-engine/src/catalogue.cpp)
target_include_directories(shv-profile-core PUBLIC src PRIVATE ../shv-engine/src)
target_link_libraries(shv-profile-core PUBLIC Symphony::KnowledgeVectorEngine shv-profile-document)
add_executable(symphony-shv-profile src/main.cpp)
target_link_libraries(symphony-shv-profile PRIVATE shv-profile-core)
foreach(target shv-profile-core symphony-shv-profile shv-profile-document)
 set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endforeach()

 list(APPEND SHV_INSTALLED "${SHV_SCHEMAS}/${name}")
