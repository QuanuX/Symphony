option(SYMPHONY_SRRAPIV_WITH_RAPI "Link an explicitly supplied licensed R API+ SDK" OFF)
set(RAPI_ROOT "" CACHE PATH "Exact extracted SDK root containing include/RApiPlus.h")
set(RAPI_PLATFORM "" CACHE STRING "Exact vendor platform lane; no automatic fallback")
add_library(srrapiv-core STATIC src/catalog.cpp src/runtime.cpp src/vendor.cpp)
target_include_directories(srrapiv-core PUBLIC src)
target_link_libraries(srrapiv-core PUBLIC Symphony::KnowledgeVectorEngine Threads::Threads)
if(SYMPHONY_SRRAPIV_WITH_RAPI)
  if(NOT IS_DIRECTORY "${RAPI_ROOT}" OR RAPI_PLATFORM STREQUAL "")
    message(FATAL_ERROR "RAPI_ROOT and RAPI_PLATFORM are required for native SDK linkage")
  endif()
  file(SHA256 "${RAPI_ROOT}/include/RApiPlus.h" HEADER_DIGEST)
  if(NOT HEADER_DIGEST STREQUAL "b996e76fbfe20c7f4d7ba2d645c45cb13d6b19df37d7522125e94e61e74df5b4")
    message(FATAL_ERROR "This mapping requires the exact reviewed R API+ 13.7.0.0 header; regenerate and qualify a separate adapter for another SDK")
  endif()
  if(APPLE AND NOT RAPI_PLATFORM MATCHES "^darwin-(10|20.6-arm64)$")
    message(FATAL_ERROR "Selected SDK lane does not match Darwin")
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT RAPI_PLATFORM MATCHES "^linux-gnu-(3.10.0|4.18)-x86_64$")
    message(FATAL_ERROR "Selected SDK lane does not match a supplied Linux x86_64 release")
  endif()
  set(VENDOR_CLOSURE "")
  foreach(name RApiPlus-optimize OmneStreamEngine-optimize OmneChannel-optimize OmneEngine-optimize _api-optimize _apipoll-stubs-optimize _kit-optimize ssl crypto)
    set(path "${RAPI_ROOT}/${RAPI_PLATFORM}/lib/lib${name}.a")
    if(NOT EXISTS "${path}")
      message(FATAL_ERROR "Selected SDK dependency absent: ${name}")
    endif()
    file(SHA256 "${path}" digest)
    string(APPEND VENDOR_CLOSURE "${name}:${digest}\n")
    target_link_libraries(srrapiv-core PUBLIC "${path}")
  endforeach()
  string(SHA256 LIBRARY_DIGEST "${VENDOR_CLOSURE}")
  target_compile_definitions(srrapiv-core PUBLIC SYMPHONY_SRRAPIV_WITH_RAPI=1 SYMPHONY_SRRAPIV_LIBRARY_SHA256="${LIBRARY_DIGEST}" SYMPHONY_SRRAPIV_PLATFORM="${RAPI_PLATFORM}")
  target_include_directories(srrapiv-core SYSTEM PRIVATE "${RAPI_ROOT}/include")
  if(APPLE)
    # Adding the SDK's usr/include through FindZLIB reorders libc++ wrappers
    # with an external LLVM compiler. The selected Apple SDK supplies -lz.
    target_link_libraries(srrapiv-core PUBLIC z ${CMAKE_DL_LIBS})
  else()
    find_package(ZLIB REQUIRED)
    target_link_libraries(srrapiv-core PUBLIC ZLIB::ZLIB ${CMAKE_DL_LIBS})
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(srrapiv-core PRIVATE LINUX _REENTRANT)
    target_link_libraries(srrapiv-core PUBLIC rt)
  endif()
endif()

endforeach()

    add_executable(srrapiv-${name}-test tests/${name}_test.cpp)
    target_link_libraries(srrapiv-${name}-test PRIVATE srrapiv-core)

    target_compile_options(srrapiv-${name}-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME srrapiv-${name} COMMAND srrapiv-${name}-test $<TARGET_FILE:symphony-srrapiv>)
  endforeach()
  add_executable(srrapiv-supervision-fixture src/main.cpp src/runtime.cpp src/catalog.cpp tests/slow_vendor.cpp)
  target_include_directories(srrapiv-supervision-fixture PRIVATE src)
  target_link_libraries(srrapiv-supervision-fixture PRIVATE Symphony::KnowledgeVectorEngine Threads::Threads)
  set_target_properties(srrapiv-supervision-fixture PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_options(srrapiv-supervision-fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME srrapiv-runtime-supervision COMMAND srrapiv-runtime-test $<TARGET_FILE:srrapiv-supervision-fixture> --slow)
