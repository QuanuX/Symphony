foreach(t sbv symphony-sbv symphony-sbv-sdk)

 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()

install(TARGETS symphony-sbv-sdk LIBRARY DESTINATION "lib/symphony/sbv-engine/${release}")
install(FILES include/symphony/sbv/sdk.h include/symphony/sbv/sdk.hpp DESTINATION "include/symphony/sbv-engine/${release}/symphony/sbv")
configure_file(cmake/SymphonySbvSdkConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/SymphonySbvSdkConfig.cmake" @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SymphonySbvSdkConfig.cmake" DESTINATION "lib/cmake/SymphonySbvSdk/${release}")
list(APPEND owned "lib/symphony/sbv-engine/${release}/${CMAKE_SHARED_LIBRARY_PREFIX}symphony-sbv-sdk${CMAKE_SHARED_LIBRARY_SUFFIX}|regular" "include/symphony/sbv-engine/${release}/symphony/sbv/sdk.h|regular" "include/symphony/sbv-engine/${release}/symphony/sbv/sdk.hpp|regular" "lib/cmake/SymphonySbvSdk/${release}/SymphonySbvSdkConfig.cmake|regular")
symphony_install_receipt_v2(COMPONENT_ID sbv-engine COMPONENT_KIND vector_engine MODULE_ID sbv-engine VECTOR_ID sbv ENGINE_ID symphony-sbv PACKAGE_ID sbv-engine VERSION "${release}" RECEIPT_PATH "${receipt}" OWNED_FILES ${owned} ENTRY_POINTS "symphony-sbv|executable|${executable}|symphony.knowledge.engine-process.v1")
set(installed_files "")
foreach(item IN LISTS owned)
 string(REGEX REPLACE "[|][^|]+$" "" path "${item}")
 list(APPEND installed_files "${path}")
endforeach()
