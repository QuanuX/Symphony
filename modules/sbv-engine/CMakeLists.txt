cmake_minimum_required(VERSION 3.30)
project(SymphonySbv VERSION 0.6.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
get_filename_component(SYMPHONY_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include("${SYMPHONY_REPOSITORY_ROOT}/cmake/SymphonyInstallReceiptV2.cmake")
if(CMAKE_CONFIGURATION_TYPES)
 message(FATAL_ERROR "SBV receipts require a single-configuration build")
endif()
if(NOT CMAKE_BUILD_TYPE)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
function(sbv_dependencies)
 set(BUILD_TESTING OFF)
 set(SYMPHONY_KVE_BUILD_TESTING OFF)
 add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/modules/sqav-databento-dbn-cpp" "${CMAKE_CURRENT_BINARY_DIR}/databento" EXCLUDE_FROM_ALL)
endfunction()
sbv_dependencies()
find_package(Threads REQUIRED)
add_library(sbv STATIC src/result.cpp src/run.cpp src/compose.cpp src/evaluate.cpp src/model.cpp src/compose_joint.cpp src/economics.cpp src/book.cpp src/liquidity.cpp src/allocation_economics.cpp)
add_library(Symphony::Sbv ALIAS sbv)
target_include_directories(sbv PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include/symphony/sbv-engine/0.6.0-dev>)
target_link_libraries(sbv PUBLIC Symphony::KnowledgeVectorEngine PRIVATE Symphony::SqavDatabentoDbn Threads::Threads)
set_target_properties(sbv PROPERTIES EXPORT_NAME Sbv)
add_executable(symphony-sbv src/main.cpp)
target_link_libraries(symphony-sbv PRIVATE sbv)
foreach(t sbv symphony-sbv)
 set_target_properties(${t} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
 target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
set(release 0.6.0-dev)
set(receipt "share/symphony/receipts/sbv-engine/${release}/install-receipt.json")
symphony_install_receipt_v2_preflight(RECEIPT_PATH "${receipt}")
set(executable "libexec/symphony/sbv-engine/${release}/symphony-sbv")
install(TARGETS symphony-sbv RUNTIME DESTINATION "libexec/symphony/sbv-engine/${release}")
set(owned "${executable}|executable")
foreach(doc INTENT MANIFEST SPEC INSTALL SKILL FEATURES)
 install(FILES "${doc}.md" DESTINATION "share/doc/symphony/sbv-engine/${release}")
 list(APPEND owned "share/doc/symphony/sbv-engine/${release}/${doc}.md|regular")
endforeach()
install(FILES OWNER-INTERFACE.json DESTINATION "share/symphony/contracts/sbv-engine/${release}")
list(APPEND owned "share/symphony/contracts/sbv-engine/${release}/OWNER-INTERFACE.json|regular")
foreach(schema admin.schema.json admin.templates.json result.schema.json pointer-stream.schema.json)
 install(FILES "schemas/v1/${schema}" DESTINATION "share/symphony/schemas/sbv-engine/${release}")
 list(APPEND owned "share/symphony/schemas/sbv-engine/${release}/${schema}|regular")
endforeach()
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/LICENSE" DESTINATION "share/licenses/symphony-sbv-engine/${release}" RENAME LICENSE-AGPL-3.0)
install(FILES "${SYMPHONY_REPOSITORY_ROOT}/libraries/knowledge-vector-engine-cpp/third_party/nlohmann/LICENSE.MIT" DESTINATION "share/licenses/symphony-sbv-engine/${release}" RENAME LICENSE-nlohmann-MIT)
list(APPEND owned "share/licenses/symphony-sbv-engine/${release}/LICENSE-AGPL-3.0|regular" "share/licenses/symphony-sbv-engine/${release}/LICENSE-nlohmann-MIT|regular")
symphony_install_receipt_v2(COMPONENT_ID sbv-engine COMPONENT_KIND vector_engine MODULE_ID sbv-engine VECTOR_ID sbv ENGINE_ID symphony-sbv PACKAGE_ID sbv-engine VERSION "${release}" RECEIPT_PATH "${receipt}" OWNED_FILES ${owned} ENTRY_POINTS "symphony-sbv|executable|${executable}|symphony.knowledge.engine-process.v1")
set(installed_files "")
foreach(item IN LISTS owned)
 string(REGEX REPLACE "[|][^|]+$" "" path "${item}")
 list(APPEND installed_files "${path}")
endforeach()
configure_file(cmake/uninstall.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake" @ONLY)
add_custom_target(uninstall-sbv-engine COMMAND ${CMAKE_COMMAND} -DINSTALL_PREFIX=${CMAKE_INSTALL_PREFIX} -P "${CMAKE_CURRENT_BINARY_DIR}/uninstall.cmake")
