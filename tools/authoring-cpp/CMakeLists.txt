cmake_minimum_required(VERSION 3.25)
project(SymphonyNativeAuthoring LANGUAGES CXX)
get_filename_component(AUTHORING_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET Symphony::KnowledgeVectorEngine)
  set(SYMPHONY_KVE_BUILD_TESTING OFF CACHE BOOL "Build foundation tests separately" FORCE)
  add_subdirectory("${AUTHORING_ROOT}/libraries/knowledge-vector-engine-cpp" "${CMAKE_CURRENT_BINARY_DIR}/knowledge-vector-engine-cpp" EXCLUDE_FROM_ALL)
endif()
add_library(symphony-authoring-support INTERFACE)
target_include_directories(symphony-authoring-support INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}"
  "${AUTHORING_ROOT}/libraries/knowledge-vector-engine-cpp/tests/support")
target_link_libraries(symphony-authoring-support INTERFACE Symphony::KnowledgeVectorEngine)
target_compile_definitions(symphony-authoring-support INTERFACE SYMPHONY_AUTHORING_ROOT="${AUTHORING_ROOT}")
function(authoring_executable target source)
  add_executable(${target} "${AUTHORING_ROOT}/${source}")
  target_link_libraries(${target} PRIVATE symphony-authoring-support)
  set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic;-Werror>)
endfunction()
