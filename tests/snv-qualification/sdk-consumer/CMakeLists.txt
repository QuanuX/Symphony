cmake_minimum_required(VERSION 3.30)
project(SnvInstalledQualification LANGUAGES CXX)
find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT REQUIRED CONFIG)
find_package(SymphonySnvCommon 0.1.0 EXACT REQUIRED CONFIG)
foreach(owner Sniv Snrv Sciv Scnv Snv SnvLocalObserver)
  find_package(Symphony${owner} 0.1.0 EXACT REQUIRED CONFIG)
  if(NOT TARGET Symphony::${owner})
    message(FATAL_ERROR "Missing exact installed ${owner} SDK")
  endif()
  get_target_property(imported Symphony::${owner} IMPORTED)
  if(NOT imported)
    message(FATAL_ERROR "Expected independently imported SDK ${owner}")
  endif()
endforeach()
add_executable(snv-installed-qualification consumer.cpp)
target_link_libraries(snv-installed-qualification PRIVATE Symphony::Sniv Symphony::Snrv Symphony::Sciv Symphony::Scnv Symphony::Snv Symphony::SnvLocalObserver)
set_target_properties(snv-installed-qualification PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
