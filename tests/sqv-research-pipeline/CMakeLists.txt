cmake_minimum_required(VERSION 3.30)
project(SymphonySqvResearchPipeline LANGUAGES CXX)
include(CTest)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(SQV_INSTALLED_ONLY "Test only independently installed packages" OFF)
if(SQV_INSTALLED_ONLY)
  find_package(SymphonySqavCapture 0.2.0 EXACT REQUIRED CONFIG)
  find_package(SymphonySqavDatabentoDbn 0.5.0 EXACT REQUIRED CONFIG)
  find_package(SymphonySqtvIntegerConversion 0.2.0 EXACT REQUIRED CONFIG)
  find_package(SymphonySqdvDelivery 0.3.0 EXACT REQUIRED CONFIG)
else()
  foreach(owner sqfv-batch-cpp sqmv-metadata-cpp sqpv-local-store-cpp sqdv-delivery-cpp sqtv-integer-conversion-cpp sqav-capture-cpp sqav-databento-dbn-cpp)
    add_subdirectory("${REPO}/modules/${owner}" "${CMAKE_CURRENT_BINARY_DIR}/${owner}")
  endforeach()
endif()
add_executable(sqv-research-pipeline pipeline.cpp)
target_link_libraries(sqv-research-pipeline PRIVATE Symphony::SqavCapture Symphony::SqtvIntegerConversion Symphony::SqdvDelivery)
set_target_properties(sqv-research-pipeline PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_options(sqv-research-pipeline PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME sqv-research-pipeline COMMAND sqv-research-pipeline)

if(NOT SQV_INSTALLED_ONLY)
  # Separate instrumented archive selection; no test hooks in shipped libraries.
  add_executable(sqv-preview-progress preview_progress.cpp "${REPO}/modules/sqdv-delivery-cpp/src/delivery.cpp")
  target_include_directories(sqv-preview-progress PRIVATE "${REPO}/modules/sqdv-delivery-cpp/include" "${REPO}/modules/sqpv-local-store-cpp/tests")
  target_link_libraries(sqv-preview-progress PRIVATE sqpv-local-store-testing Symphony::KnowledgeVectorEngine)
  set_target_properties(sqv-preview-progress PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_options(sqv-preview-progress PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME sqv-preview-progress COMMAND sqv-preview-progress)
endif()

if(NOT SQV_INSTALLED_ONLY)
  configure_file("${REPO}/modules/sqav-databento-dbn-cpp/tests/public_fixture.hpp" "${CMAKE_CURRENT_BINARY_DIR}/public_fixture.hpp" COPYONLY)
endif()
add_executable(sqv-nonlive-pipeline nonlive.cpp)
target_include_directories(sqv-nonlive-pipeline PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(sqv-nonlive-pipeline PRIVATE Symphony::SqavDatabentoDbn Symphony::SqdvDelivery)
set_target_properties(sqv-nonlive-pipeline PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_options(sqv-nonlive-pipeline PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME sqv-nonlive-pipeline COMMAND sqv-nonlive-pipeline)
