cmake_minimum_required(VERSION 3.30)
project(InstalledSbvConsumers LANGUAGES C CXX)
include(CTest)

set(SBV_PREFIX "" CACHE PATH "Exact independently installed SBV prefix")
if(NOT IS_ABSOLUTE "${SBV_PREFIX}" OR NOT IS_DIRECTORY "${SBV_PREFIX}")
  message(FATAL_ERROR "Supply an existing absolute SBV_PREFIX")
endif()
file(REAL_PATH "${SBV_PREFIX}" selected_prefix)
set(release "0.33.0-dev")

function(require_selected_path path)
  if(NOT IS_ABSOLUTE "${path}" OR NOT EXISTS "${path}")
    message(FATAL_ERROR "Installed SBV path is absent or relative: ${path}")
  endif()
  file(REAL_PATH "${path}" canonical)
  cmake_path(IS_PREFIX selected_prefix "${canonical}" NORMALIZE selected)
  if(NOT selected)
    message(FATAL_ERROR "Installed SBV path escapes selected prefix: ${canonical}")
  endif()
endfunction()

# These installed contracts use exact versioned configuration directories and
# release variables, rather than CMake ConfigVersion files.
foreach(sdk SbvSdk SbvProvider SbvInterop)
  set(package "Symphony${sdk}")
  set(config "${selected_prefix}/lib/cmake/${package}/${release}")
  require_selected_path("${config}/${package}Config.cmake")
  set(${package}_DIR "${config}")
  find_package(${package} CONFIG REQUIRED PATHS "${config}" NO_DEFAULT_PATH)
  file(REAL_PATH "${${package}_DIR}" selected_config)
  if(NOT selected_config STREQUAL config OR
     NOT "${${package}_VERSION}" STREQUAL release)
    message(FATAL_ERROR "Installed SBV configuration identity differs: ${package}")
  endif()
  set(target "Symphony::${sdk}")
  get_target_property(imported "${target}" IMPORTED)
  if(NOT imported)
    message(FATAL_ERROR "Expected imported SBV target: ${target}")
  endif()
  get_target_property(includes "${target}" INTERFACE_INCLUDE_DIRECTORIES)
  if(NOT includes)
    message(FATAL_ERROR "Installed SBV target has no public headers: ${target}")
  endif()
  foreach(path IN LISTS includes)
    require_selected_path("${path}")
  endforeach()
  if(sdk STREQUAL "SbvSdk")
    get_target_property(library "${target}" IMPORTED_LOCATION)
    require_selected_path("${library}")
  endif()
endforeach()

add_executable(sbv-installed-c-consumer sdk_consumer.c)
target_link_libraries(sbv-installed-c-consumer PRIVATE Symphony::SbvSdk)
set_target_properties(sbv-installed-c-consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
add_executable(sbv-installed-cpp-consumer sdk_consumer.cpp)
target_link_libraries(sbv-installed-cpp-consumer PRIVATE Symphony::SbvSdk)
set_target_properties(sbv-installed-cpp-consumer PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
add_executable(sbv-installed-provider-abi provider_abi_test.c)
target_link_libraries(sbv-installed-provider-abi PRIVATE Symphony::SbvProvider)
set_target_properties(sbv-installed-provider-abi PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
add_executable(sbv-installed-interop interop_consumer.cpp)
target_link_libraries(sbv-installed-interop PRIVATE Symphony::SbvInterop)
set_target_properties(sbv-installed-interop PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)

add_test(NAME sbv-installed-sdk-correspondence
  COMMAND "${CMAKE_COMMAND}"
    "-DC_CONSUMER=$<TARGET_FILE:sbv-installed-c-consumer>"
    "-DCXX_CONSUMER=$<TARGET_FILE:sbv-installed-cpp-consumer>"
    "-DREQUEST_DIRECTORY=${CMAKE_CURRENT_BINARY_DIR}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/verify_sdk.cmake")
add_test(NAME sbv-installed-provider-abi COMMAND sbv-installed-provider-abi)
add_test(NAME sbv-installed-interop COMMAND sbv-installed-interop)
