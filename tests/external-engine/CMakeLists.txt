cmake_minimum_required(VERSION 3.31)
project(BetaEmpiricalPriceExample VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT IS_ABSOLUTE "${BETA_SDK_PREFIX}")
 message(FATAL_ERROR "Select an absolute, complete installed public SDK prefix")
endif()
set(BETA_EXAMPLE_VERSION "0.1.0" CACHE STRING "Example's independent version")
if(NOT BETA_EXAMPLE_VERSION MATCHES "^0[.]1[.][01]$")
 message(FATAL_ERROR "Only the two qualified compatible example versions are admitted")
endif()
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT CONFIG REQUIRED
 PATHS "${BETA_SDK_PREFIX}/lib/cmake/SymphonyKnowledgeVectorEngine-0.2.0-dev" NO_DEFAULT_PATH)
find_package(SymphonySbvProvider CONFIG REQUIRED
 PATHS "${BETA_SDK_PREFIX}/lib/cmake/SymphonySbvProvider/0.33.0-dev" NO_DEFAULT_PATH)
find_package(SymphonySbvSdk CONFIG REQUIRED
 PATHS "${BETA_SDK_PREFIX}/lib/cmake/SymphonySbvSdk/0.33.0-dev" NO_DEFAULT_PATH)
if(NOT SymphonySbvProvider_VERSION STREQUAL "0.33.0-dev" OR NOT SymphonySbvSdk_VERSION STREQUAL "0.33.0-dev")
 message(FATAL_ERROR "Exact installed SBV 0.33.0-dev interfaces required")
endif()
foreach(dependency Symphony::KnowledgeVectorEngine Symphony::SbvProvider Symphony::SbvSdk)
 get_target_property(is_imported ${dependency} IMPORTED)
 if(NOT is_imported)
  message(FATAL_ERROR "Only installed imported SDK targets are admitted")
 endif()
endforeach()
add_executable(beta-empirical-price engine.cpp)
target_link_libraries(beta-empirical-price PRIVATE Symphony::KnowledgeVectorEngine)
add_library(beta-empirical-provider MODULE provider.cpp)
target_link_libraries(beta-empirical-provider PRIVATE Symphony::KnowledgeVectorEngine Symphony::SbvProvider)
set_target_properties(beta-empirical-provider PROPERTIES PREFIX "" SUFFIX ".so" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
add_executable(beta-sdk-client sdk-client.cpp)
target_link_libraries(beta-sdk-client PRIVATE Symphony::KnowledgeVectorEngine Symphony::SbvSdk)
add_executable(beta-provider-probe provider-probe.cpp)
target_link_libraries(beta-provider-probe PRIVATE Symphony::KnowledgeVectorEngine Symphony::SbvProvider ${CMAKE_DL_LIBS})
foreach(target beta-empirical-price beta-empirical-provider beta-sdk-client beta-provider-probe)
 target_compile_definitions(${target} PRIVATE BETA_EXAMPLE_VERSION="${BETA_EXAMPLE_VERSION}")
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
# This is an independent qualification example, not a new Symphony owner or an
# automatic registry admission. Install only into a private example prefix.
install(TARGETS beta-empirical-price RUNTIME DESTINATION "bin/beta-empirical-price/${BETA_EXAMPLE_VERSION}")
install(TARGETS beta-empirical-provider LIBRARY DESTINATION "lib/beta-empirical-price/${BETA_EXAMPLE_VERSION}")
add_executable(beta-fixture fixture.cpp)
target_compile_options(beta-fixture PRIVATE -Wall -Wextra -Wpedantic -Werror)
