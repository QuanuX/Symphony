cmake_minimum_required(VERSION 3.30)
project(SymphonySqvAdministrationTests LANGUAGES CXX)
include(CTest)
get_filename_component(SYMPHONY_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(module sqmv-metadata-engine sqfv-flow-engine sqtv-conversion-engine sqpv-inspection-engine sqdv-checkpoint-engine sqav-attempt-engine)
  add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/modules/${module}" "${CMAKE_CURRENT_BINARY_DIR}/${module}")
endforeach()
# Fixture writers remain test-only; production inspection links no Store writer.
function(fixture_dependencies)
  set(BUILD_TESTING OFF)
  add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/modules/sqdv-delivery-cpp" "${CMAKE_CURRENT_BINARY_DIR}/delivery" EXCLUDE_FROM_ALL)
  add_subdirectory("${SYMPHONY_REPOSITORY_ROOT}/modules/sqav-databento-dbn-cpp" "${CMAKE_CURRENT_BINARY_DIR}/databento" EXCLUDE_FROM_ALL)
endfunction()
fixture_dependencies()
if(BUILD_TESTING)
  foreach(binding "sqmv-metadata-engine|metadata_validate" "sqfv-flow-engine|flow_validate" "sqtv-conversion-engine|conversion_validate")
    string(REPLACE "|" ";" parts "${binding}")
    list(GET parts 0 module)
    list(GET parts 1 operation)
    add_executable(${module}-test validation_test.cpp)
    target_link_libraries(${module}-test PRIVATE ${module}-core)
    set_target_properties(${module}-test PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    target_compile_options(${module}-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME ${operation} COMMAND ${module}-test "${operation}" "${SYMPHONY_REPOSITORY_ROOT}/modules/${module}/tests/fixtures/${operation}.json")
  endforeach()
  add_executable(sqv-admin-fixtures fixtures.cpp)
  target_include_directories(sqv-admin-fixtures PRIVATE "${SYMPHONY_REPOSITORY_ROOT}/tools/sqv-administration-cpp")
  target_link_libraries(sqv-admin-fixtures PRIVATE Symphony::SqdvDelivery Symphony::SqavDatabentoDbn Symphony::KnowledgeVectorEngine)
  set_target_properties(sqv-admin-fixtures PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_options(sqv-admin-fixtures PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

  add_executable(sqv-reader-test reader_test.cpp)
  target_link_libraries(sqv-reader-test PRIVATE sqpv-inspection-engine-core Symphony::SqpvLocalStore)
  set_target_properties(sqv-reader-test PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  target_compile_options(sqv-reader-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME read_only_store_boundary COMMAND sqv-reader-test)
endif()
