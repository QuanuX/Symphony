cmake_minimum_required(VERSION 3.31)
project(SbvLaunchFixtures LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SymphonySbvProvider CONFIG REQUIRED PATHS "${SBV_PREFIX}/lib/cmake/SymphonySbvProvider/0.33.0-dev" NO_DEFAULT_PATH)
if(NOT SymphonySbvProvider_VERSION STREQUAL "0.33.0-dev")
  message(FATAL_ERROR "Exact installed SBV provider interface required")
endif()
find_path(JSON_INCLUDE nlohmann/json.hpp PATHS "${SBV_PREFIX}/include/symphony/knowledge-vector-engine/0.2.0-dev" NO_DEFAULT_PATH REQUIRED)
add_library(launch-feedback MODULE feedback_fixture.cpp)
target_link_libraries(launch-feedback PRIVATE Symphony::SbvProvider)
target_include_directories(launch-feedback PRIVATE "${JSON_INCLUDE}")
set_target_properties(launch-feedback PROPERTIES PREFIX "" SUFFIX ".so")
add_executable(launch-data fixture.cpp)
target_include_directories(launch-data PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
