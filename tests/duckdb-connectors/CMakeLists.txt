cmake_minimum_required(VERSION 3.25)
project(SymphonyInstalledDuckDBConsumers LANGUAGES CXX)
enable_testing()
find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT REQUIRED CONFIG
  PATHS "${DUCKDB_SDK_PREFIX}" NO_DEFAULT_PATH)
get_target_property(imported Symphony::KnowledgeVectorEngine IMPORTED)
if(NOT imported)
  message(FATAL_ERROR "Installed consumer requires the imported foundation")
endif()
foreach(target scv-process shv-process recovery)
  if(target STREQUAL "scv-process")
    set(source scv-connector-test.cpp)
  elseif(target STREQUAL "shv-process")
    set(source shv-connector-test.cpp)
  else()
    set(source recovery.cpp)
  endif()
  add_executable(${target} "${source}")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(${target} PRIVATE Symphony::KnowledgeVectorEngine ${CMAKE_DL_LIBS})
  set_target_properties(${target} PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
endforeach()
if(APPLE)
  set(library libduckdb.dylib)
else()
  set(library libduckdb.so)
endif()
foreach(domain scv shv)
  if(domain STREQUAL "scv")
    set(version 0.2.0-dev)
  else()
    set(version 0.4.0-dev)
  endif()
  set(directory "${DUCKDB_INSTALL_PREFIX}/libexec/symphony/${domain}-graph-duckdb-connector/${version}")
  add_test(NAME ${domain}-installed-process COMMAND ${domain}-process
    --engine "${directory}/symphony-${domain}-graph-duckdb-connector"
    --library "${directory}/${library}" --version "${version}")
  add_test(NAME ${domain}-installed-recovery COMMAND recovery --domain ${domain}
    --engine "${directory}/symphony-${domain}-graph-duckdb-connector" --version "${version}"
    --fault-engine "${DUCKDB_BUILD_ROOT}/${domain}-graph-duckdb-connector/${domain}-graph-duckdb-commit-fault-test"
    --retain-root "${DUCKDB_RETAINED_ROOT}/${domain}")
endforeach()
set_tests_properties(scv-installed-process shv-installed-process scv-installed-recovery shv-installed-recovery
  PROPERTIES TIMEOUT 300)
