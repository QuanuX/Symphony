cmake_minimum_required(VERSION 3.30)
project(InstalledMechanicsConsumer LANGUAGES CXX)
find_package(SymphonyKnowledgeVectorEngine 0.2.0 EXACT REQUIRED CONFIG)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Symphony::KnowledgeVectorEngine)
if(SDK_INCLUDE_SNV_COMMON)
  find_package(SymphonySnvCommon 0.1.0 EXACT REQUIRED CONFIG)
  target_link_libraries(consumer PRIVATE Symphony::SnvCommon)
  target_compile_definitions(consumer PRIVATE SDK_INCLUDE_SNV_COMMON=1)
endif()
set_target_properties(consumer PROPERTIES CXX_STANDARD 26 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
